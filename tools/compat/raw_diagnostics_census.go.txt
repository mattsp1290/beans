package vault
import("encoding/json";"os";"path/filepath";"strings";"testing";"regexp";"reflect";"github.com/mattsp1290/beans/issue";"github.com/mattsp1290/beans/gitops")
func TestMigrationRawDiagnostics(t *testing.T) {
 type node struct { Path []int; Kind string; Data []int; Mode uint32 }
 type input struct { Name []int; Action,Kind string }
 type row struct { Input input; Error []int; Tree []node }
 ints:=func(s string)[]int {v:=make([]int,len(s));for i:=range v{v[i]=int(s[i])};return v}
 str:=func(v []int)string {b:=make([]byte,len(v));for i,n:=range v {b[i]=byte(n)};return string(b)}
 snapshot:=func(root string)[]node {out:=[]node{};e:=filepath.WalkDir(root,func(path string,d os.DirEntry,e error)error {if e!=nil{return e};if path==root{return nil};info,e:=os.Lstat(path);if e!=nil{return e};n:=node{Path:ints(strings.TrimPrefix(path,root+"/")),Data:[]int{},Mode:uint32(info.Mode().Perm())};switch {case info.Mode()&os.ModeSymlink!=0:n.Kind="symlink";s,e:=os.Readlink(path);if e!=nil{return e};n.Data=ints(s);case info.IsDir():n.Kind="directory";default:n.Kind="file";b,e:=os.ReadFile(path);if e!=nil{return e};n.Data=ints(string(b))};out=append(out,n);return nil});if e!=nil{t.Fatal(e)};return out}
 var names []string;for b:=0;b<256;b++{names=append(names,"cfg"+string([]byte{byte(b)})+"x.toml")};names=append(names,"雪.toml","café.toml","🌱.toml","bad\xff.xyz","bad\xfe.yaml","bad\xff.yml")
 var inputs []input
 for _,name:=range names {for _,action:=range []string{"user","project","hub","workflow"} {for _,kind:=range []string{"missing","directory","malformed","vocabulary"} {inputs=append(inputs,input{ints(name),action,kind})}}}
 for _,name:=range []string{"normal","bad\xff","bad\xfe","nul\x00","雪","with\nnewline","nested/file"} {for _,kind:=range []string{"missing","directory","existing","symlink","parent-file"} {inputs=append(inputs,input{ints(name),"write",kind})}}
 for _,name:=range []string{"normal","bad\xff","bad\xfe","nul\x00","雪","with\nnewline"} {for _,kind:=range []string{"hub-malformed","hub-directory","project-malformed"} {inputs=append(inputs,input{ints(name),"index",kind})};inputs=append(inputs,input{ints(name),"dirs","projects-file"},input{ints(name),"reload","missing"})}
 var rows []row;tempRe:=regexp.MustCompile(`\.bn-write-[0-9]+`)
 for _,in:=range inputs {root:=t.TempDir();path:=filepath.Join(root,str(in.Name));if !strings.ContainsRune(path,0) {if e:=os.MkdirAll(filepath.Dir(path),0755);e!=nil{t.Fatal(e)};switch in.Kind {
 case "directory":if e:=os.Mkdir(path,0755);e!=nil{t.Fatal(e)}
 case "malformed":if e:=os.WriteFile(path,[]byte("name=["),0644);e!=nil{t.Fatal(e)}
 case "vocabulary":if e:=os.WriteFile(path,[]byte("[workflow]\ndefault='absent'\n"),0644);e!=nil{t.Fatal(e)}
 case "existing":if e:=os.WriteFile(path,[]byte("old user data"),0644);e!=nil{t.Fatal(e)}
 case "symlink":if e:=os.WriteFile(filepath.Join(root,"target"),[]byte("unrelated target"),0644);e!=nil{t.Fatal(e)};target,e:=filepath.Rel(filepath.Dir(path),filepath.Join(root,"target"));if e!=nil{t.Fatal(e)};if e=os.Symlink(target,path);e!=nil{t.Fatal(e)}
 case "parent-file":if e:=os.MkdirAll(path,0755);e!=nil{t.Fatal(e)};if e:=os.WriteFile(filepath.Join(path,"blocker"),[]byte("user file"),0644);e!=nil{t.Fatal(e)};path=filepath.Join(path,"blocker","leaf")
 }}
 if !strings.ContainsRune(path,0) {switch in.Kind {
case "hub-malformed","hub-directory","project-malformed","projects-file":if e:=os.MkdirAll(path,0755);e!=nil{t.Fatal(e)};config:=filepath.Join(path,"beans.toml");if in.Kind=="hub-directory" {if e:=os.Mkdir(config,0755);e!=nil{t.Fatal(e)}} else if in.Kind=="projects-file" {if e:=os.WriteFile(filepath.Join(path,"projects"),[]byte("user file"),0644);e!=nil{t.Fatal(e)}} else {if in.Kind=="project-malformed" {config=filepath.Join(path,"projects/p/beans.toml");if e:=os.MkdirAll(filepath.Dir(config),0755);e!=nil{t.Fatal(e)}};if e:=os.WriteFile(config,[]byte("name=["),0644);e!=nil{t.Fatal(e)}}
}}
 before:=snapshot(root);var e error
 switch in.Action {case "user":_,e=issue.LoadUserConfig(path);case "project":_,e=issue.LoadProjectConfig(path);case "hub":_,e=issue.LoadHubConfig(path);case "workflow":_,e=issue.LoadWorkflow(path,nil,nil);case "write":e=gitops.WriteFile(path,[]byte("replacement\xff\n"));case "index":_,e=Load(path);case "dirs":_,e=ProjectDirs(path);case "reload":var ix *Index;ix,e=Load(filepath.Join(root,"live"));if e==nil {e=ix.Reload(path)}}
 out:=row{Input:in,Error:[]int{}};if e!=nil{out.Error=ints(tempRe.ReplaceAllString(strings.ReplaceAll(e.Error(),root+"/",""),".bn-write-TEMP"))};after:=snapshot(root);if in.Action=="write" {out.Tree=after} else if !reflect.DeepEqual(before,after) {t.Fatal("read mutated tree")};rows=append(rows,out)
 }
 b,e:=json.Marshal(map[string]any{"cases":rows});if e!=nil{t.Fatal(e)};if e=os.WriteFile(os.Getenv("BN_RAW_DIAGNOSTICS_OUTPUT"),b,0600);e!=nil{t.Fatal(e)}
}
