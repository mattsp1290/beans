func indexBytes(s string) []int {
	v := make([]int, len(s))
	for i := 0; i < len(s); i++ {
		v[i] = int(s[i])
	}
	return v
}
func indexStrings(v []string) any {
	if v == nil {
		return nil
	}
	r := [][]int{}
	for _, s := range v {
		r = append(r, indexBytes(s))
	}
	return r
}
func indexLinks(v []LinkRef) any {
	r := []any{}
	for _, l := range v {
		r = append(r, map[string]any{"from": indexBytes(l.From), "to": indexBytes(l.To), "kind": l.Kind})
	}
	return r
}
func indexWF(v issue.WorkflowConfig) any {
	return map[string]any{"statuses": v.Statuses, "default": v.Default, "active": v.Active, "terminal": v.Terminal, "transitions": v.Transitions}
}
func indexSnapshot(ix *Index, root string) any {
	if ix == nil {
		return nil
	}
	notes := []any{}
	for _, n := range ix.order {
		id := any(nil)
		aliases := noteAliases(n)
		if aliases == nil {
			aliases = []string{}
		}
		description, body := "", n.docBody
		switch n.Kind {
		case KindIssue:
			id = indexBytes(n.Issue.ID)
			description = n.Issue.Description
			body = n.Issue.Body
		case KindRequest:
			id = indexBytes(n.Request.ID)
			body = n.Request.Body
		case KindMemory:
			body = n.Memory.Body
		case KindHandoff:
			id = indexBytes(n.Handoff.ID)
			body = n.Handoff.Body
		case KindPlan:
			id = indexBytes(n.Plan.ID)
			body = n.Plan.Body
		}
		raw := []any{}
		for _, l := range n.rawOut {
			raw = append(raw, map[string]any{"target": indexBytes(l.to), "kind": l.kind})
		}
		notes = append(notes, map[string]any{"kind": n.Kind, "path": indexBytes(n.Path), "basename": indexBytes(n.Basename), "project": indexBytes(n.Project), "title": indexBytes(n.Title), "tags": indexStrings(n.Tags), "id": id, "aliases": indexStrings(aliases), "description": indexBytes(description), "body": indexBytes(body), "raw": raw, "outlinks": indexLinks(n.Outlinks)})
	}
	keys := func(m map[string]*Note) []string {
		v := []string{}
		for k := range m {
			v = append(v, k)
		}
		sort.Strings(v)
		return v
	}
	owners := []any{}
	for _, k := range keys(ix.Notes) {
		owners = append(owners, map[string]any{"basename": indexBytes(k), "path": indexBytes(ix.Notes[k].Path)})
	}
	aliases := []any{}
	ak := []string{}
	for k := range ix.Aliases {
		ak = append(ak, k)
	}
	sort.Strings(ak)
	for _, k := range ak {
		aliases = append(aliases, map[string]any{"alias": indexBytes(k), "basename": indexBytes(ix.Aliases[k])})
	}
	backlinks := []any{}
	bk := []string{}
	for k := range ix.Backlinks {
		bk = append(bk, k)
	}
	sort.Strings(bk)
	for _, k := range bk {
		backlinks = append(backlinks, map[string]any{"basename": indexBytes(k), "links": indexLinks(ix.Backlinks[k])})
	}
	warnings := []any{}
	for _, w := range ix.Warnings {
		warnings = append(warnings, map[string]any{"path": indexBytes(w.Path), "error": strings.ReplaceAll(w.Err.Error(), root, "{ROOT}")})
	}
	assets := [][]int{}
	as := []string{}
	for p := range ix.Assets {
		as = append(as, p)
	}
	sort.Strings(as)
	for _, p := range as {
		assets = append(assets, indexBytes(p))
	}
	projects := []any{}
	pk := []string{}
	for p := range ix.Projects {
		pk = append(pk, p)
	}
	sort.Strings(pk)
	for _, p := range pk {
		v := ix.Projects[p]
		projects = append(projects, map[string]any{"name": indexBytes(p), "prefix": indexBytes(v.Config.Prefix), "workflow": indexWF(v.Workflow)})
	}
	return map[string]any{"notes": notes, "owners": owners, "aliases": aliases, "backlinks": backlinks, "warnings": warnings, "assets": assets, "projects": projects, "workflow": indexWF(ix.Workflow)}
}

type indexEntry struct {
	Path []int
	Kind string
	Data []int
}

func indexFiles(t *testing.T, root string) []indexEntry {
	t.Helper()
	out := []indexEntry{}
	e := filepath.WalkDir(root, func(p string, d fs.DirEntry, e error) error {
		if e != nil {
			return e
		}
		if p == root {
			return nil
		}
		rel, _ := filepath.Rel(root, p)
		kind := "file"
		data := []byte{}
		if d.IsDir() {
			kind = "directory"
		} else if d.Type()&os.ModeSymlink != 0 {
			kind = "symlink"
			v, e := os.Readlink(p)
			if e != nil {
				return e
			}
			data = []byte(v)
		} else {
			v, e := os.ReadFile(p)
			if e != nil {
				return e
			}
			data = v
		}
		out = append(out, indexEntry{indexBytes(rel), kind, indexBytes(string(data))})
		return nil
	})
	if e != nil {
		t.Fatal(e)
	}
	return out
}
func TestMigrationIndex(t *testing.T) {
	rows := []any{}
	seed := func(root string) {
		e := filepath.WalkDir("testdata/hub", func(p string, d fs.DirEntry, e error) error {
			if e != nil {
				return e
			}
			rel, _ := filepath.Rel("testdata/hub", p)
			full := filepath.Join(root, rel)
			if d.IsDir() {
				return os.MkdirAll(full, 0755)
			}
			data, e := os.ReadFile(p)
			if e != nil {
				return e
			}
			return os.WriteFile(full, data, 0644)
		})
		if e != nil {
			t.Fatal(e)
		}
	}
	write := func(root, p, s string) {
		p = filepath.Join(root, p)
		if e := os.MkdirAll(filepath.Dir(p), 0755); e != nil {
			t.Fatal(e)
		}
		if e := os.WriteFile(p, []byte(s), 0644); e != nil {
			t.Fatal(e)
		}
	}
	link := func(root, p, target string) {
		full := filepath.Join(root, p)
		if e := os.MkdirAll(filepath.Dir(full), 0755); e != nil {
			t.Fatal(e)
		}
		if e := os.Symlink(target, full); e != nil {
			t.Fatal(e)
		}
	}
	planFile := string(plan.Scaffold("a-plan-a123", "test", time.Date(2026, 1, 1, 0, 0, 0, 0, time.UTC)))
	planRoot := "projects/a/plans/a-plan-a123-test"
	for scenario := 0; scenario < 26; scenario++ {
		rootName := "hub"
		if scenario >= 22 {
			rootName = []string{"hub[", "hub*", "hub?", "hub\\"}[scenario-22]
		}
		root := filepath.Join(t.TempDir(), rootName)
		if e := os.Mkdir(root, 0755); e != nil {
			t.Fatal(e)
		}
		seed(root)
		switch scenario {
		case 1:
			write(root, "docs/same.md", "---\naliases: [shared-schema, alias]\ntags: [x, 1]\n---\n# Doc\n[[a-open001]] ![[missing]]")
			write(root, "projects/b/docs/same.md", "# Later")
			write(root, "docs/other.md", "[[alias]]")
		case 2:
			write(root, "projects/.hidden/beans.toml", "name='hidden'\nprefix='hidden'\n")
			write(root, "projects/.hidden/docs/hidden.md", "# hidden")
			write(root, "templates/a.md", "# skip")
			write(root, "docs/.hidden.md", "# skip")
		case 3:
			link(root, "projects/sym", "a")
			link(root, "docs/sym.md", "../projects/a/issues/a-open001.md")
			link(root, "docs/missing.md", "absent")
			link(root, "docs/dead.png", "absent")
		case 4:
			write(root, "beans.toml", "[workflow]\ndefault='missing'\n")
		case 5:
			write(root, "projects/zero/beans.toml", "[")
		case 6:
			write(root, planRoot+"/plan.md", planFile)
		case 7:
			write(root, planRoot+"/plan.md", "broken")
		case 8:
			write(root, "projects/a/plans/plain", "not directory")
			link(root, "projects/a/plans/link", "plain")
		case 9:
			write(root, "projects/a/plans/.a-plan-a123-test.backup/plan.md", planFile)
		case 10:
			write(root, planRoot+"/plan.md", planFile)
			write(root, planRoot+"/plan.md.tmp", "unfinished")
		case 11:
			write(root, "projects/unknown/plans/a-plan-a123-test/plan.md", planFile)
		case 12:
			write(root, "projects/b/plans/a-plan-a123-test/plan.md", planFile)
		case 13:
			write(root, "projects/a/plans/.a-plan-a123-test.backup/plan.md", planFile)
			write(root, planRoot+"/plan.md", planFile)
			write(root, "projects/a/plans/.stage/keep", "stage")
		case 14:
			write(root, "projects/a/handoffs/a-handoff1-live.md", "---\nid: a-handoff1\ntitle: Live\ncreated: 2026-01-01T00:00:00Z\nupdated: 2026-01-01T00:00:00Z\nissue: '[[a-open001]]'\n---\n[[parity]]\n")
			write(root, "projects/a/handoffs/archive/2026/a-handoff2-old.md", "---\nid: a-handoff2\ntitle: Old\ncreated: 2026-01-01T00:00:00Z\nupdated: 2026-01-01T00:00:00Z\n---\nold")
			write(root, "projects/a/requests/a-req1.md", "---\nid: a-req1\naliases: [reqalias]\ntitle: Request\nstatus: open\npriority: 2\ncreated: 2026-01-01T00:00:00Z\nupdated: 2026-01-01T00:00:00Z\nissues: ['[[a-open001]]']\n---\n[[parity]]\n## Log\nopaque\n")
		case 15:
			write(root, "docs/raw\xff.md", "---\ntitle: Name\n---\nraw\xff [[parity]]")
			p := filepath.Join(root, "projects/a/issues/a-open001.md")
			v, _ := os.ReadFile(p)
			write(root, "projects/a/issues/a-open001.md", string(v)+"raw\xff [[parity]]\n")
		}
		if scenario >= 16 && scenario <= 21 {
			write(root, "docs/same.md", "# Before\n[[a-open001]]")
			write(root, "projects/b/docs/same.md", "# Later")
			write(root, planRoot+"/plan.md", planFile)
		}
		initial := indexFiles(t, root)
		ix, e := Load(root)
		errorText := ""
		if e != nil {
			errorText = strings.ReplaceAll(e.Error(), root, "{ROOT}")
		}
		stages := []any{map[string]any{"error": errorText, "index": indexSnapshot(ix, root), "files": indexFiles(t, root)}}
		steps := []any{}
		if scenario >= 16 && scenario <= 21 && ix != nil {
			changes := []indexEntry{}
			paths := []string{}
			all := false
			switch scenario {
			case 16:
				changes = append(changes, indexEntry{indexBytes("docs/same.md"), "file", indexBytes("# Changed\n[[missing]]")})
				paths = []string{"docs/same.md"}
			case 17:
				changes = append(changes, indexEntry{indexBytes("docs/same.md"), "delete", nil})
				paths = []string{"docs/same.md"}
			case 18:
				changes = append(changes, indexEntry{indexBytes("beans.toml"), "file", indexBytes("[")})
				paths = []string{"beans.toml"}
			case 19:
				changes = append(changes, indexEntry{indexBytes(planRoot + "/plan.md"), "file", indexBytes("broken")})
				paths = []string{planRoot + "/sections/change.md"}
			case 20:
				changes = append(changes, indexEntry{indexBytes(planRoot), "delete", nil})
				paths = []string{planRoot + "/plan.md"}
			case 21:
				paths = []string{"docs/same.md", "../outside"}
			}
			for _, change := range changes {
				p := stringFromInts(change.Path)
				if change.Kind == "delete" {
					if e := os.RemoveAll(filepath.Join(root, p)); e != nil {
						t.Fatal(e)
					}
				} else {
					write(root, p, stringFromInts(change.Data))
				}
			}
			pathsBytes := [][]int{}
			for _, p := range paths {
				pathsBytes = append(pathsBytes, indexBytes(p))
			}
			steps = append(steps, map[string]any{"changes": changes, "paths": pathsBytes, "all": all})
			e = ix.Reload(paths...)
			errorText = ""
			if e != nil {
				errorText = strings.ReplaceAll(e.Error(), root, "{ROOT}")
			}
			stages = append(stages, map[string]any{"error": errorText, "index": indexSnapshot(ix, root), "files": indexFiles(t, root)})
		}
		rows = append(rows, map[string]any{"root": rootName, "initial": initial, "steps": steps, "stages": stages})
	}
	patterns := []any{}
	for _, p := range []string{"*", "?", "a*", "a?", "[a-z]", "[^a]", "[]", "[a-]", "[", "x*[", "a\\*", "[\\]]", "[é-界]", "*?*", "*/*", "\\", "[\xff]", "**a"} {
		for _, n := range []string{"", "a", "b", "a*", "aa", "é", "界", "/", "a/b", "\xff", "\xfe", "�"} {
			ok, e := filepath.Match(p, n)
			patterns = append(patterns, map[string]any{"pattern": indexBytes(p), "name": indexBytes(n), "match": ok, "error": e != nil})
		}
	}
	data, e := json.Marshal(map[string]any{"cases": rows, "patterns": patterns})
	if e != nil {
		t.Fatal(e)
	}
	if e := os.WriteFile(os.Getenv("BN_INDEX_OUTPUT"), data, 0600); e != nil {
		t.Fatal(e)
	}
}
func stringFromInts(v []int) string {
	b := make([]byte, len(v))
	for i, n := range v {
		b[i] = byte(n)
	}
	return string(b)
}
