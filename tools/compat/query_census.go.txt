// Lists here deliberately capture semantic identities with nonnil empty
// arrays. Transport nilness is qualified separately at the CLI/API boundary.
func queryIDs(v []*issue.Issue) []string {
	out := []string{}
	for _, i := range v {
		out = append(out, i.ID)
	}
	return out
}
func queryPaths(v []*Note) [][]int {
	out := [][]int{}
	for _, n := range v {
		out = append(out, indexBytes(n.Path))
	}
	return out
}
func querySnapshot(ix *Index) any {
	scopes := []any{}
	for _, p := range []string{"", "a", "b", "missing", "raw\xff"} {
		for _, all := range []bool{false, true} {
			blocked := []any{}
			for _, b := range ix.Blocked(p, all) {
				blocked = append(blocked, map[string]any{"id": b.Issue.ID, "blockers": b.Blockers})
			}
			nodes, edges := ix.Graph(p, all)
			if edges == nil {
				edges = []GraphEdge{}
			}
			scopes = append(scopes, map[string]any{"project": indexBytes(p), "all": all, "ready": queryIDs(ix.Ready(p, all)), "blocked": blocked, "nodes": nodes, "edges": edges})
		}
	}
	projects := []any{}
	for _, p := range []string{"", "a", "b", "missing", "raw\xff"} {
		for _, archive := range []bool{false, true} {
			hs := []string{}
			for _, h := range ix.ProjectHandoffs(p, archive) {
				hs = append(hs, h.ID)
			}
			plans := []string{}
			for _, v := range ix.ProjectPlans(p) {
				plans = append(plans, v.ID)
			}
			projects = append(projects, map[string]any{"project": indexBytes(p), "archived": archive, "issues": queryIDs(ix.ProjectIssues(p, archive)), "handoffs": hs, "plans": plans})
		}
	}
	refs := []any{}
	for _, r := range []string{"a-open001", "b-blocked001", "a-epic001", "a-child001", "b-child001", "shared-schema", "[[ a-open001#anchor|label ]]", "missing", "[[a-missing999]]", "docs/a-open001.md", "\xff", "[[ ]]", "\u2003a-open001\u2003"} {
		target, i, ok := ix.ResolveIssueRef(r)
		id := ""
		if ok {
			id = i.ID
		}
		refs = append(refs, map[string]any{"ref": indexBytes(r), "target": indexBytes(target), "issue": id, "children": queryIDs(ix.Children(r)), "parents": queryIDs(ix.Parents(r)), "backlinks": indexLinks(ix.IssueBacklinks(r, false)), "all_backlinks": indexLinks(ix.IssueBacklinks(r, true))})
	}
	blockers := []any{}
	for _, i := range ix.ProjectIssues("", true) {
		resolved, missing := ix.Blockers(i)
		if missing == nil {
			missing = []string{}
		}
		blockers = append(blockers, map[string]any{"id": i.ID, "resolved": queryIDs(resolved), "unresolved": missing})
	}
	searches := []any{}
	for _, q := range []string{"", "  ", "parity", "a-open001", "schema", "raw\xff", "\xff", "needle-only-in-section", "label", "request", "handoff", "İ", "ſ", "Σ", "\u2003PARITY\u2003", "not-found"} {
		for _, kinds := range [][]Kind{nil, {KindIssue}, {KindDoc, KindMemory}, {KindRequest, KindPlan, KindHandoff}, {"unknown"}, {KindIssue, KindIssue}} {
			for _, archive := range []bool{false, true} {
				hits := ix.SearchWithOptions(q, SearchOptions{Kinds: kinds, IncludeArchivedHandoffs: archive})
				if hits == nil {
					hits = []Hit{}
				}
				// Byte arrays retain invalid filename bytes that Go JSON would replace.
				rows := []any{}
				for _, h := range hits {
					rows = append(rows, map[string]any{"kind": h.Kind, "id": indexBytes(h.ID), "basename": indexBytes(h.Basename), "title": indexBytes(h.Title), "project": indexBytes(h.Project), "path": indexBytes(h.Path), "score": h.Score})
				}
				searches = append(searches, map[string]any{"q": indexBytes(q), "kinds": kinds, "archived": archive, "hits": rows})
			}
		}
	}
	requests := []any{}
	for _, f := range []RequestFilter{{}, {Project: "a"}, {Project: "b"}, {Status: "resolved"}, {Status: "declined"}, {Status: "hold"}, {Label: "label"}, {Priority: new(int)}, {Query: "REQUEST"}, {Query: "\u2003raw\xff\u2003"}, {Query: "actor"}, {Query: "label"}, {Terminal: true}, {Project: "a", Terminal: true, Priority: func() *int { v := 2; return &v }()}} {
		ids := []string{}
		for _, r := range ix.ProjectRequests(f.Project, f.Status, f.Label, f.Priority, f.Query, f.Terminal) {
			ids = append(ids, r.ID)
		}
		requests = append(requests, map[string]any{"filter": f, "ids": ids})
	}
	cycles := ix.Cycles()
	if cycles == nil {
		cycles = [][]string{}
	}
	executions := []any{}
	for _, p := range ix.ProjectPlans("") {
		v, ok := ix.PlanExecution(p.ID)
		executions = append(executions, map[string]any{"id": p.ID, "ok": ok, "report": v})
	}
	_, missingOK := ix.PlanExecution("missing")
	return map[string]any{"executions": executions, "missing_execution": missingOK, "scopes": scopes, "projects": projects, "refs": refs, "blockers": blockers, "searches": searches, "requests": requests, "cycles": cycles}
}

	if e := os.WriteFile(os.Getenv("BN_QUERY_OUTPUT"), data, 0600); e != nil {

	}
}

type RequestFilter struct {
	Project, Status, Label string
	Priority               *int
	Query                  string
	Terminal               bool
}

func planScaffoldForQuery(dir string) error {
	if e := plan.WriteScaffold(dir, "a-plan-a123", "test", time.Date(2026, 1, 1, 0, 0, 0, 0, time.UTC)); e != nil {
		return e
	}
	p := filepath.Join(dir, "plan.md")
	v, e := os.ReadFile(p)
	if e != nil {
		return e
	}
	s := strings.Replace(string(v), "created:", "sections: [sections/one.md]\ncreated:", 1)
	if e := os.MkdirAll(filepath.Join(dir, "sections"), 0755); e != nil {
		return e
	}
	if e := os.WriteFile(filepath.Join(dir, "sections/one.md"), []byte("needle-only-in-section\n"), 0644); e != nil {
		return e
	}
	return os.WriteFile(p, []byte(s), 0644)
}
