package vault

import (
	"encoding/json"
	"os"
	"sort"
	"testing"
	"time"
)

func TestMigrationIndexBench(t *testing.T) {
	root := os.Getenv("BN_INDEX_BENCH_HUB")
	samples := []float64{}
	for i := 0; i < 23; i++ {
		start := time.Now()
		ix, e := Load(root)
		elapsed := float64(time.Since(start)) / float64(time.Millisecond)
		if e != nil {
			t.Fatal(e)
		}
		active := 0
		for _, iss := range ix.Issues {
			if !iss.Archived {
				active++
			}
		}
		if active != 5000 || len(ix.Issues) != 5001 {
			t.Fatalf("loaded %d issues", len(ix.Issues))
		}
		if i >= 3 {
			samples = append(samples, elapsed)
		}
	}
	ordered := append([]float64(nil), samples...)
	sort.Float64s(ordered)
	record := map[string]any{"schema": "beans-index-load-v1", "issue_count": 5000, "total_issue_count": 5001, "warmup_runs": 3, "samples_ms": samples, "median_ms": (ordered[9] + ordered[10]) / 2, "p95_ms": ordered[19], "min_ms": ordered[0], "max_ms": ordered[19]}
	data, e := json.Marshal(record)
	if e != nil {
		t.Fatal(e)
	}
	if e := os.WriteFile(os.Getenv("BN_INDEX_BENCH_OUTPUT"), data, 0600); e != nil {
		t.Fatal(e)
	}
}
