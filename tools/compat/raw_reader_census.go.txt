func readerInts(b []byte) []int {
	v := make([]int, len(b))
	for i, b := range b {
		v[i] = int(b)
	}
	return v
}
func TestMigrationRawReader(t *testing.T) {
	heads := map[string]string{
		"issue":   "id: p-one\ntitle: Title # keep\ntype: task\nstatus: open\npriority: 2\n",
		"request": "id: p-r-one\naliases: [p-r-one]\ntitle: Title # keep\nstatus: open\npriority: 2\n",
		"memory":  "key: key\ntype: reference # keep\n",
		"handoff": "id: p-one\ntitle: Title # keep\n",
	}
	stamp := "created: 2026-01-01T00:00:00Z\nupdated: 2026-01-01T00:00:00Z\n"
	rows := []any{}
	add := func(kind, prefix string, padding int, sequence []byte, suffix string) {
		path := "projects/p/issues/p-one.md"
		switch kind {
		case "request":
			path = "projects/p/requests/p-r-one.md"
		case "memory":
			path = "projects/p/memories/key.md"
		case "handoff":
			path = "projects/p/handoffs/p-one.md"
		}
		fm := []byte(prefix + strings.Repeat("x", padding))
		fm = append(fm, sequence...)
		fm = append(fm, []byte(suffix)...)
		input := append([]byte("---\n"), fm...)
		input = append(input, []byte("---\nbody\xff\n")...)
		row := map[string]any{"kind": kind, "path": path, "prefix": prefix, "padding": padding, "sequence": readerInts(sequence), "suffix": suffix}
		outputs := map[string]any{}
		for _, mode := range []string{"noop", "first", "last"} {
			var out []byte
			var e error
			now := time.Date(2027, 1, 1, 0, 0, 0, 0, time.UTC)
			switch kind {
			case "issue":
				d, err := Parse(path, input)
				e = err
				if e == nil {
					if mode == "first" {
						d.Title = "Changed"
					}
					if mode == "last" {
						d.Updated = now
					}
					out, e = Encode(d)
				}
			case "request":
				d, err := ParseRequest(path, input)
				e = err
				if e == nil {
					if mode == "first" {
						d.Title = "Changed"
					}
					if mode == "last" {
						d.Updated = now
					}
					out, e = EncodeRequest(d)
				}
			case "memory":
				d, err := ParseMemory(path, input)
				e = err
				if e == nil {
					if mode == "first" {
						d.Type = "feedback"
					}
					if mode == "last" {
						d.Updated = now
					}
					out, e = EncodeMemory(d)
				}
			case "handoff":
				d, err := ParseHandoff(path, input)
				e = err
				if e == nil {
					if mode == "first" {
						d.Title = "Changed"
					}
					if mode == "last" {
						d.Updated = now
					}
					out, e = EncodeHandoff(d)
				}
			}
			text := ""
			if e != nil {
				text = e.Error()
			}
			result := map[string]any{"error": text}
			if e == nil {
				reread, readError := readerRead(kind, path, out)
				message := ""
				if readError != nil {
					message = readError.Error()
				} else if !bytes.Equal(reread, out) {
					t.Fatal("Go reread changes encoded output")
				}
				result["reread_error"] = message
				if mode == "noop" {
					if !bytes.Equal(out, input) {
						t.Fatal("Go no-op changes input")
					}
				} else {
					result["bytes"] = readerInts(out)
				}
			}
			outputs[mode] = result
		}
		row["outputs"] = outputs
		rows = append(rows, row)
	}
	for _, kind := range []string{"issue", "request", "memory", "handoff"} {
		base := heads[kind] + stamp
		for b := 0; b < 256; b++ {
			for _, target := range []int{0, 511, 512, 1023} {
				prefix := base + "# "
				padding := target - len(prefix)
				if padding < 0 {
					padding = 0
				}
				add(kind, prefix, padding, []byte{byte(b)}, "\n")
			}
		}
		for _, sequence := range [][]byte{{255}, {192, 175}, {226, 130}, {237, 160, 128}, {240, 128, 128, 175}, {244, 144, 128, 128}, {248}, {254}, {0}, {31}, {239, 191, 190}, {239, 187, 191}, {226, 130, 172}} {
			for _, target := range []int{0, 510, 511, 512, 513, 1023, 1024, 1025, 4096} {
				for _, extra := range []string{"# ", "bad: [}\n# ", "...\nkey: value\n# ", "...\ntrailer\n# "} {
					prefix := base + extra
					padding := target - len(prefix)
					if padding < 0 {
						padding = 0
					}
					add(kind, prefix, padding, sequence, "\n")
				}
			}
		}
		for _, sequence := range [][]byte{{255}, {192, 175}, {226, 130}, {237, 160, 128}, {0}, {226, 130, 172}} {
			for _, target := range []int{0, 511, 512, 513, 1023, 1024, 4096} {
				for _, prefix := range []string{"bad: [}\npadding: '", "bad: [}\n# "} {
					padding := target - len(prefix)
					if padding < 0 {
						padding = 0
					}
					suffix := "\n" + base
					if strings.Contains(prefix, "padding:") {
						suffix = "'\n" + base
					}
					add(kind, prefix, padding, sequence, suffix)
				}
			}
		}

	}
	data, e := json.Marshal(map[string]any{"cases": rows})
	if e != nil {
		t.Fatal(e)
	}
	if e := os.WriteFile(os.Getenv("BN_RAW_READER_OUTPUT"), data, 0600); e != nil {
		t.Fatal(e)
	}
	fmt.Println("raw reader cases", len(rows))
}

func readerRead(kind, path string, input []byte) ([]byte, error) {
	switch kind {
	case "issue":
		d, e := Parse(path, input)
		if e != nil {
			return nil, e
		}
		return Encode(d)
	case "request":
		d, e := ParseRequest(path, input)
		if e != nil {
			return nil, e
		}
		return EncodeRequest(d)
	case "memory":
		d, e := ParseMemory(path, input)
		if e != nil {
			return nil, e
		}
		return EncodeMemory(d)
	case "handoff":
		d, e := ParseHandoff(path, input)
		if e != nil {
			return nil, e
		}
		return EncodeHandoff(d)
	}
	panic("unknown kind")
}
func TestMigrationRawReaderRead(t *testing.T) {
	raw, e := os.ReadFile(os.Getenv("BN_RUST_RAW_READER_INPUT"))
	if e != nil {
		t.Fatal(e)
	}
	var rows []struct {
		Kind, Path, ExpectedError string
		Bytes                     []int
	}
	if e := json.Unmarshal(raw, &rows); e != nil {
		t.Fatal(e)
	}
	accepted, rejected := 0, 0
	for i, r := range rows {
		input := make([]byte, len(r.Bytes))
		for j, b := range r.Bytes {
			input[j] = byte(b)
		}
		output, e := readerRead(r.Kind, r.Path, input)
		message := ""
		if e != nil {
			message = e.Error()
		}
		if message != r.ExpectedError {
			t.Fatalf("%d: got %s want %s", i, message, r.ExpectedError)
		}
		if e == nil {
			accepted++
			if !bytes.Equal(input, output) {
				t.Fatalf("Go rewrites Rust output %d", i)
			}
		} else {
			rejected++
		}
	}
	t.Logf("Go read %d actual Rust outputs: %d accepted, %d matching rejections", len(rows), accepted, rejected)
}
