func wfBytes(s string) []int {
	out := make([]int, len(s))
	for i, b := range []byte(s) {
		out[i] = int(b)
	}
	return out
}
func wfView(w WorkflowFile) any {
	return map[string]any{"statuses": w.Statuses, "default": w.Default, "active": w.Active, "terminal": w.Terminal, "transitions": w.Transitions}
}
func wfConfig(w WorkflowConfig) any {
	return map[string]any{"statuses": w.Statuses, "default": w.Default, "active": w.Active, "terminal": w.Terminal, "transitions": w.Transitions}
}
func wfError(e error) string {
	if e == nil {
		return ""
	}
	return e.Error()
}

			file = "explicit.yaml"
		}
		path := filepath.Join(work, file)
		os.WriteFile(path, []byte(explicit), 0644)
		w, e := LoadWorkflow(path, []byte("[workflow]\nunknown=1"), []byte("[workflow]\nunknown=1"))

			os.Mkdir(path, 0755)
		case "dangling-link":
			os.Symlink("absent.toml", path)
		case "parent-file":
			os.WriteFile(filepath.Join(work, "parent"), []byte("x"), 0644)

		case "symlink":
			os.WriteFile(filepath.Join(work, "target.toml"), []byte("[workflow]\ndefault=\"in_progress\""), 0644)

		}
		w, e := LoadWorkflow(path, nil, nil)

	}

func TestMigrationWorkflowFileCrossRead(t *testing.T) {
	data, e := os.ReadFile(os.Getenv("BN_WORKFLOW_RUST_INPUT"))
	if e != nil {
		t.Fatal(e)
	}
	var inputs []struct {
		Name  string `json:"name"`
		Kind  string `json:"kind"`
		Bytes []int  `json:"bytes"`
	}
	if e = json.Unmarshal(data, &inputs); e != nil {
		t.Fatal(e)
	}
	rows := []any{}
	for _, row := range inputs {
		if row.Kind != "project" {
			continue
		}
		data := make([]byte, len(row.Bytes))
		for i, b := range row.Bytes {
			data[i] = byte(b)
		}
		file, e := decodeWorkflowFile("writer.toml", data)
		rows = append(rows, map[string]any{"name": "project-writer-" + row.Name, "file_config": wfView(file.Workflow), "error": wfError(e)})
	}
	out, e := json.Marshal(map[string]any{"reads": rows})
	if e != nil {
		t.Fatal(e)
	}
	if e = os.WriteFile(os.Getenv("BN_WORKFLOW_FILE_OUTPUT"), out, 0644); e != nil {
		t.Fatal(e)
	}
}
