// AST census only: never execute upstream target expectations.
package main

import (
	"crypto/sha256"
	"encoding/json"
	"fmt"
	"go/ast"
	"go/parser"
	"go/token"
	"os"
	"strconv"
	"strings"
)

func integer(e ast.Expr) int {
	switch x := e.(type) {
	case *ast.BasicLit:
		v, err := strconv.Atoi(x.Value)
		if err != nil {
			panic(err)
		}
		return v
	case *ast.BinaryExpr:
		if x.Op == token.MUL {
			return integer(x.X) * integer(x.Y)
		}
	}
	panic("unsupported integer AST")
}
func inputValue(e ast.Expr) string {
	switch x := e.(type) {
	case *ast.BasicLit:
		v, err := strconv.Unquote(x.Value)
		if err != nil {
			panic(err)
		}
		return v
	case *ast.BinaryExpr:
		if x.Op == token.ADD {
			return inputValue(x.X) + inputValue(x.Y)
		}
	case *ast.CallExpr:
		if sel, ok := x.Fun.(*ast.SelectorExpr); ok && sel.X.(*ast.Ident).Name == "strings" && sel.Sel.Name == "Repeat" {
			return strings.Repeat(inputValue(x.Args[0]), integer(x.Args[1]))
		}
	}
	panic("unsupported input AST")
}
func main() {
	rows := []any{}
	hashes := map[string]string{}
	for _, spec := range []struct{ file, table string }{{"decode_test.go", "unmarshalTests"}, {"node_test.go", "nodeTests"}} {
		raw, e := os.ReadFile(os.Args[1] + "/" + spec.file)
		if e != nil {
			panic(e)
		}
		hashes[spec.file] = fmt.Sprintf("%x", sha256.Sum256(raw))
		fs := token.NewFileSet()
		f, e := parser.ParseFile(fs, spec.file, raw, parser.ParseComments)
		if e != nil {
			panic(e)
		}
		found := false
		ast.Inspect(f, func(n ast.Node) bool {
			v, ok := n.(*ast.ValueSpec)
			if !ok {
				return true
			}
			for _, name := range v.Names {
				if name.Name != spec.table {
					continue
				}
				found = true
				table := v.Values[0].(*ast.CompositeLit)
				for i, item := range table.Elts {
					row := item.(*ast.CompositeLit)
					literal := row.Elts[0]
					input := inputValue(literal)
					mode := "decode"
					annotation := ""
					for _, prefix := range []string{"[encode]", "[decode]"} {
						if strings.HasPrefix(input, prefix) {
							annotation = prefix
							input = strings.TrimPrefix(input, prefix)
							if prefix == "[encode]" {
								mode = "encode-only"
							}
						}
					}
					bytes := []int{}
					for _, b := range []byte(input) {
						bytes = append(bytes, int(b))
					}
					target := row.Elts[1]
					a, b := fs.Position(target.Pos()).Offset, fs.Position(target.End()).Offset
					rows = append(rows, map[string]any{"id": fmt.Sprintf("%s/%04d", spec.table, i), "table": spec.table, "index": i, "file": spec.file, "line": fs.Position(row.Pos()).Line, "target_expression": string(raw[a:b]), "source_expression": string(raw[fs.Position(literal.Pos()).Offset:fs.Position(literal.End()).Offset]), "input": bytes, "mode": mode, "annotation": annotation})
				}
				return false
			}
			return true
		})
		if !found {
			panic("missing upstream table")
		}
	}
	json.NewEncoder(os.Stdout).Encode(map[string]any{"rows": rows, "source_sha256": hashes})
}
