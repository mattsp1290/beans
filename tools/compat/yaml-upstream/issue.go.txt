func ycObserve(c ycCase) map[string]any {
	if c.EditField != "" {
		return ycEdit(c)
	}

	input := ycInput(c)
	var model any
	var out []byte
	var e error
	var pe error
	switch c.Kind {
	case "issue":
		var d *Issue
		d, e = Parse(c.Path, input)
		pe = e
		if e == nil {
			model = ycModel(d)
			out, e = Encode(d)
			if c.Edit {
				d.Status = "in_progress"
				out, e = Encode(d)
				if e == nil {
					d, e = Parse(c.Path, out)
					pe = e
					if e == nil {
						model = ycModel(d)
					}
				}
			}
		}
	case "request":
		var d *Request
		d, e = ParseRequest(c.Path, input)
		pe = e
		if e == nil {
			model = ycModel(d)
			out, e = EncodeRequest(d)
		}
	case "memory":
		var d *Memory
		d, e = ParseMemory(c.Path, input)
		pe = e
		if e == nil {
			model = ycModel(d)
			out, e = EncodeMemory(d)
		}
	case "handoff":
		var d *Handoff
		d, e = ParseHandoff(c.Path, input)
		pe = e
		if e == nil {
			model = ycModel(d)
			out, e = EncodeHandoff(d)
		}
	case "workflow":
		var d workflowFile
		d, e = decodeWorkflowFile(c.Path, input)
		pe = e
		if e == nil {
			model = ycWorkflow(d.Workflow)
		}
	default:
		panic(c.Kind)
	}
	return ycResult(model, out, pe, e)
}

// Supplemental generated edits retain the initial model and no-op separately,
// and retain encoded bytes even when the historical edit cannot be reparsed.
func ycEdit(c ycCase) map[string]any {
	initial := c
	initial.EditField = ""
	r := map[string]any{"initial_read": ycObserve(initial), "edit_bytes": nil, "edit_error": []int{}, "reread": nil, "owned_range": nil, "unchanged_ranges": nil, "unknown_ranges": nil}
	d, e := Parse(c.Path, ycInput(c))
	if e != nil {
		return r
	}
	start, end := 4, 4
	lineOffset := func(line int) int {
		n := 4
		for _, value := range d.orig.fmLines[:line] {
			n += len(value) + 1
		}
		return n
	}
	unknown := [][]int{}
	for _, span := range d.orig.spans {
		if span.key == "status" {
			start, end = lineOffset(span.start), lineOffset(span.end)
		}
		if !ownedKeySet[span.key] {
			unknown = append(unknown, []int{lineOffset(span.start), lineOffset(span.end)})
		}
	}
	if c.EditField == "description" {
		start = lineOffset(len(d.orig.fmLines)) + 4
		end = start + len(d.Description)
	}
	r["owned_range"] = []int{start, end}
	r["unknown_ranges"] = unknown
	if c.EditField == "status" {
		d.Status = "in_progress"
	} else {
		SetDescription(d, "Upstream qualification replacement.\nSecond line.")
	}
	out, e := Encode(d)
	if e != nil {
		r["edit_error"] = ycBytes(e.Error())
		return r
	}
	r["edit_bytes"] = ycBytes(string(out))
	initial.Input = ycBytes(string(out))
	r["reread"] = ycObserve(initial)
	replacementEnd := end + len(out) - len(ycInput(c))
	ranges := [][]int{{0, start, 0, start}, {end, len(ycInput(c)), replacementEnd, len(out)}}
	r["unchanged_ranges"] = ranges
	if _, err := Parse(c.Path, out); err == nil {
		for _, span := range ranges {
			if !bytes.Equal(ycInput(c)[span[0]:span[1]], out[span[2]:span[3]]) {
				panic("valid edit changed bytes outside owned range")
			}
		}
		for _, span := range unknown {
			if !(span[1] <= start || span[0] >= end) {
				panic("valid edit overlaps unknown YAML")
			}
		}
	}
	return r
}

func TestMigrationYAMLCensus(t *testing.T) {
	rows := map[string]any{}
	for _, c := range ycReadCases(t) {
		if c.Kind == "manifest" || c.Kind == "graph" {
			continue
		}
		r := ycObserve(c)
		if c.EditField != "" {
			rows[c.ID] = r
			continue
		}
		if c.Edit {
			initial := c
			initial.Edit = false
			r["initial_read"] = ycObserve(initial)
		}
		if out, ok := r["noop"].([]int); ok {
			c.Input = out
			c.Edit = false
			r["reread"] = ycObserve(c)
		} else {
			r["reread"] = nil
		}
		rows[c.ID] = r
	}
	ycSave(t, rows)
}

func ycWorkflow(w WorkflowFile) any {
	m := ycValue(reflect.ValueOf(w)).(map[string]any)
	if w.Transitions != nil {
		keys := []string{}
		for k := range w.Transitions {
			keys = append(keys, k)
		}
		sort.Strings(keys)
		a := []any{}
		for _, k := range keys {
			a = append(a, map[string]any{"key": ycBytes(k), "value": ycValue(reflect.ValueOf(w.Transitions[k]))})
		}
		m["transitions"] = a
	}
	return m
}
