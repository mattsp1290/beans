func ycBytes(s string) []int {
	v := make([]int, len(s))
	for i, b := range []byte(s) {
		v[i] = int(b)
	}
	return v
}
func ycName(s string) string {
	switch s {
	case "ID":
		return "id"
	case "URL":
		return "url"
	case "SHA":
		return "sha"
	case "BlockedBy":
		return "blocked_by"
	case "RequestedBy":
		return "requested_by"
	case "AffectedAreas":
		return "affected_areas"
	case "ExecutionOrder":
		return "execution_order"
	case "ChangeGraph":
		return "change_graph"
	case "SectionBodies":
		return "section_bodies"
	}
	return strings.ToLower(s)
}
func ycValue(v reflect.Value) any {
	if v.Kind() == reflect.Interface || v.Kind() == reflect.Pointer {
		if v.IsNil() {
			return nil
		}
		return ycValue(v.Elem())
	}
	if v.Type() == reflect.TypeOf(time.Time{}) {
		t := v.Interface().(time.Time)
		_, z := t.Zone()
		return map[string]any{"seconds": t.Unix(), "nanoseconds": t.Nanosecond(), "offset_seconds": z}
	}
	switch v.Kind() {
	case reflect.String:
		return ycBytes(v.String())
	case reflect.Slice, reflect.Array:
		if v.Kind() == reflect.Slice && v.IsNil() {
			return nil
		}
		a := []any{}
		for i := 0; i < v.Len(); i++ {
			a = append(a, ycValue(v.Index(i)))
		}
		return a
	case reflect.Map:
		if v.IsNil() {
			return nil
		}
		m := map[string]any{}
		it := v.MapRange()
		for it.Next() {
			m[it.Key().String()] = ycValue(it.Value())
		}
		return m
	case reflect.Struct:
		m := map[string]any{}
		for i := 0; i < v.NumField(); i++ {
			f := v.Type().Field(i)
			if !f.IsExported() {
				continue
			}
			if f.Name == "Extra" {
				m["unknown"] = ycExtra(v.Field(i).Interface().(yaml.Node))
				continue
			}
			if f.Name == "Path" {
				continue
			}
			m[ycName(f.Name)] = ycValue(v.Field(i))
		}
		return m
	default:
		return v.Interface()
	}
}

type ycCase struct {
	ID, Kind, Path string
	EditField      string
	Input          []int
	Edit           bool
}

func ycReadCases(t *testing.T) []ycCase {
	b, e := os.ReadFile(os.Getenv("BN_YAML_INPUT"))
	if e != nil {
		t.Fatal(e)
	}
	var c []ycCase
	if e = json.Unmarshal(b, &c); e != nil {
		t.Fatal(e)
	}
	return c
}
func ycInput(c ycCase) []byte {
	b := make([]byte, len(c.Input))
	for i, x := range c.Input {
		b[i] = byte(x)
	}
	return b
}
func ycResult(model any, out []byte, pe, e error) map[string]any {
	r := map[string]any{"error": []int{}, "parse_error": []int{}, "encode_error": []int{}, "model": model, "noop": nil}
	if pe != nil {
		r["parse_error"] = ycBytes(pe.Error())
		r["error"] = ycBytes(pe.Error())
	} else if e != nil {
		r["encode_error"] = ycBytes(e.Error())
		r["error"] = ycBytes(e.Error())
	} else if out != nil {
		r["noop"] = ycBytes(string(out))
	}
	return r
}

func ycSave(t *testing.T, rows map[string]any) {
	b, e := json.Marshal(rows)
	if e != nil {
		t.Fatal(e)
	}
	if e = os.WriteFile(os.Getenv("BN_YAML_OUTPUT"), b, 0600); e != nil {
		t.Fatal(e)
	}
}

func ycNode(n *yaml.Node) any {
	k := map[yaml.Kind]string{yaml.ScalarNode: "scalar", yaml.SequenceNode: "sequence", yaml.MappingNode: "mapping", yaml.AliasNode: "alias"}
	m := map[string]any{"kind": ycBytes(k[n.Kind]), "line": n.Line, "tag": ycBytes(n.Tag), "anchor": ycBytes(n.Anchor)}
	if n.Kind == yaml.AliasNode {
		m["anchor"] = ycBytes(n.Value)
	}
	if n.Kind == yaml.ScalarNode {
		m["value"] = ycBytes(n.Value)
		m["null"] = n.Tag == "!!null"
	}
	if len(n.Content) > 0 {
		a := []any{}
		for _, c := range n.Content {
			a = append(a, ycNode(c))
		}
		m["children"] = a
	}
	return m
}
func ycExtra(n yaml.Node) any {
	a := []any{}
	for i := 0; i+1 < len(n.Content); i += 2 {
		a = append(a, map[string]any{"key": ycBytes(n.Content[i].Value), "value": ycNode(n.Content[i+1])})
	}
	return a
}
func ycModel(d any) any {
	m := ycValue(reflect.ValueOf(d)).(map[string]any)
	m["path"] = ycBytes(reflect.ValueOf(d).Elem().FieldByName("Path").String())
	for _, key := range []string{"aliases", "labels", "blocked_by", "issues", "tags", "log", "sections"} {
		if x, ok := m[key]; ok && x == nil {
			m[key] = []any{}
		}
	}
	return m
}
