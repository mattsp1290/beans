func ycObserve(c ycCase) map[string]any {
	input := ycInput(c)
	var model any
	var out []byte
	var e error
	var pe error
	if c.Kind == "manifest" {
		var d *Plan
		d, e = Parse(c.Path, input)
		pe = e
		if e == nil {
			model = ycModel(d)
			m := model.(map[string]any)
			m["path"] = ycBytes(d.Path)
			m["section_bodies"] = []any{}
			out, e = Encode(d)
		}
	} else {
		var d ChangeGraph
		d, e = ParseGraph(c.Path, string(input))
		pe = e
		model = ycValue(reflect.ValueOf(d))
	}
	return ycResult(model, out, pe, e)
}

func TestMigrationYAMLCensus(t *testing.T) {
	rows := map[string]any{}
	for _, c := range ycReadCases(t) {
		if c.Kind != "manifest" && c.Kind != "graph" {
			continue
		}
		r := ycObserve(c)
		if out, ok := r["noop"].([]int); ok {
			c.Input = out
			c.Edit = false
			r["reread"] = ycObserve(c)
		} else {
			r["reread"] = nil
		}
		rows[c.ID] = r
	}
	ycSave(t, rows)
}
