package issue

import (
	"bytes"
	"encoding/json"
	"fmt"
	"os"
	"testing"
	"time"
)

func rawInts(b []byte) []int {
	v := make([]int, len(b))
	for i, c := range b {
		v[i] = int(c)
	}
	return v
}
func TestMigrationRawCodecs(t *testing.T) {
	rows := []any{}
	stamp := "created: 2026-01-01T00:00:00Z\nupdated: 2026-01-01T00:00:00Z\n"
	for _, kind := range []string{"issue", "request", "memory", "handoff"} {
		path := "projects/p/issues/p-one.md"
		head := "id: p-one\ntitle: Title\ntype: task\nstatus: open\npriority: 2\n"
		switch kind {
		case "request":
			path = "projects/p/requests/p-r-one.md"
			head = "id: p-r-one\naliases: [p-r-one]\ntitle: Title\nstatus: open\npriority: 2\n"
		case "memory":
			path = "projects/p/memories/key.md"
			head = "key: key\ntype: reference\n"
		case "handoff":
			path = "projects/p/handoffs/p-one.md"
			head = "id: p-one\ntitle: Title\n"
		}
		bodies := [][]byte{}
		for i := 0; i < 256; i++ {
			b := []byte(fmt.Sprintf("desc%c\n## Body\nbody%c\n## Log\n- opaque%c\n  continuation%c\n\n## Tail\ntail%c\n", i, i, i, i, i))
			bodies = append(bodies, b)
		}
		for _, b := range []string{"", "raw\xff", "raw\xe2\x82", "raw\xc0\xaf\n", "raw\xf0\x80\x80\xaf\n", "```\n## Log\nraw\xff\n```\n## Log\n- opaque\xff", "```\n## Log\n- opaque\xff", "raw\xff\n## Log\n- opaque\xff\n\t\u2003\n## Tail\nraw\xff"} {
			bodies = append(bodies, []byte(b))
		}
		for n, body := range bodies {
			input := append([]byte("---\n"+head+stamp+"unknown: {keep: yes} # preserved\n---\n"), body...)
			replacement := append([]byte("replacement"), body...)
			replacement = append(replacement, []byte("\nline\n")...)
			row := map[string]any{"kind": kind, "path": path, "input": rawInts(input), "replacement": rawInts(replacement), "name": fmt.Sprintf("%s-%d", kind, n)}
			outputs := map[string]any{}
			for _, mutation := range []string{"noop", "metadata", "body", "body_raw", "append"} {
				var out []byte
				var e error
				entry := LogEntry{At: time.Date(2026, 2, 1, 0, 0, 0, 0, time.UTC), Actor: "actor", Event: "event"}
				switch kind {
				case "issue":
					var d *Issue
					d, e = Parse(path, input)
					if e == nil {
						switch mutation {
						case "metadata":
							d.Title = "Changed"
						case "body":
							SetDescription(d, "replacement\nline")
						case "body_raw":
							SetDescription(d, string(replacement))
						case "append":
							AppendLog(d, entry)
						}
						out, e = Encode(d)
					}
				case "request":
					var d *Request
					d, e = ParseRequest(path, input)
					if e == nil {
						switch mutation {
						case "metadata":
							d.Title = "Changed"
						case "body":
							SetRequestBody(d, "replacement\nline")
						case "body_raw":
							SetRequestBody(d, string(replacement))
						case "append":
							AppendRequestLog(d, entry)
						}
						out, e = EncodeRequest(d)
					}
				case "memory":
					var d *Memory
					d, e = ParseMemory(path, input)
					if e == nil {
						switch mutation {
						case "metadata":
							d.Type = "feedback"
						case "body":
							d.Body = "replacement\nline\n"
						case "body_raw":
							d.Body = string(replacement)
						}
						out, e = EncodeMemory(d)
					}
				case "handoff":
					var d *Handoff
					d, e = ParseHandoff(path, input)
					if e == nil {
						switch mutation {
						case "metadata":
							d.Title = "Changed"
						case "body":
							d.Body = "replacement\nline\n"
						case "body_raw":
							d.Body = string(replacement)
						}
						out, e = EncodeHandoff(d)
					}
				}
				errorText := ""
				if e != nil {
					errorText = e.Error()
				}
				outputs[mutation] = map[string]any{"bytes": rawInts(out), "error": errorText}
			}
			row["outputs"] = outputs
			rows = append(rows, row)
		}
	}
	data, e := json.Marshal(map[string]any{"cases": rows})
	if e != nil {
		t.Fatal(e)
	}
	if e := os.WriteFile(os.Getenv("BN_RAW_CODEC_OUTPUT"), data, 0600); e != nil {
		t.Fatal(e)
	}
}

func TestMigrationRawCodecRead(t *testing.T) {
	raw, e := os.ReadFile(os.Getenv("BN_RUST_RAW_CODEC_INPUT"))
	if e != nil {
		t.Fatal(e)
	}
	var rows []struct {
		Name, Kind, Path string
		Bytes            []int
	}
	if e := json.Unmarshal(raw, &rows); e != nil {
		t.Fatal(e)
	}
	for _, r := range rows {
		input := make([]byte, len(r.Bytes))
		for i, b := range r.Bytes {
			input[i] = byte(b)
		}
		var output []byte
		var e error
		switch r.Kind {
		case "issue":
			d, err := Parse(r.Path, input)
			e = err
			if e == nil {
				output, e = Encode(d)
			}
		case "request":
			d, err := ParseRequest(r.Path, input)
			e = err
			if e == nil {
				output, e = EncodeRequest(d)
			}
		case "memory":
			d, err := ParseMemory(r.Path, input)
			e = err
			if e == nil {
				output, e = EncodeMemory(d)
			}
		case "handoff":
			d, err := ParseHandoff(r.Path, input)
			e = err
			if e == nil {
				output, e = EncodeHandoff(d)
			}
		default:
			t.Fatal("unknown kind")
		}
		if e != nil {
			t.Fatalf("%s: %v", r.Name, e)
		}
		if !bytes.Equal(input, output) {
			t.Fatalf("Go rewrites Rust output %s", r.Name)
		}
	}
	t.Logf("fixed Go accepted and roundtripped %d actual Rust raw-byte outputs", len(rows))
}
