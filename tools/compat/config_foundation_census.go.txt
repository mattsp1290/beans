func configBytes(s string) []int {
	out := make([]int, len(s))
	for i, b := range []byte(s) {
		out[i] = int(b)
	}
	return out
}
func configList(xs []string) any {
	if xs == nil {
		return nil
	}
	out := []any{}
	for _, x := range xs {
		out = append(out, configBytes(x))
	}
	return out
}
func configInput(w WorkflowConfig) any {
	var transitions any
	if w.Transitions != nil {
		rows := []any{}
		for from, tos := range w.Transitions {
			rows = append(rows, map[string]any{"from": configBytes(from), "to": configList(tos)})
		}
		transitions = rows
	}
	return map[string]any{"statuses": configList(w.Statuses), "default": configBytes(w.Default), "active": configList(w.Active), "terminal": configList(w.Terminal), "transitions": transitions}
}
func configView(w WorkflowConfig) any {
	return map[string]any{"statuses": w.Statuses, "default": w.Default, "active": w.Active, "terminal": w.Terminal, "transitions": w.Transitions}
}
func configFileInput(w WorkflowFile) any {
	return configInput(WorkflowConfig{Statuses: w.Statuses, Default: w.Default, Active: w.Active, Terminal: w.Terminal, Transitions: w.Transitions})
}
func TestMigrationConfigFoundation(t *testing.T) {
	rows := []any{}
	merges := []any{}
	types := []any{}
	durations := []any{}
	values := []string{"", "open", "closed", " open ", "雪", "🌱", "true", "\n", "\t", "\u0085", "\u2003", "\u200b", "\u202e", "\ufeff", "a\x00b", "bad\nname", string([]byte{255}), "�", string([]byte{226, 130}), "\ue000", "\U0010ffff", " \xff \u2003", " \xe2\x82 \u0085", "\x7f", "\x1c"}
	add := func(name string, w WorkflowConfig) {
		message := ""
		if e := w.Validate(); e != nil {
			message = e.Error()
		}
		queries := append([]string{"open", "in_progress", "ready_for_review", "ready_for_validation", "ready_for_merge", "blocked", "closed", "done", "ghost"}, values...)
		classes := []any{}
		for _, s := range queries {
			classes = append(classes, map[string]any{"status": configBytes(s), "valid": w.IsValid(s), "active": w.IsActive(s), "terminal": w.IsTerminal(s), "hold": w.IsHold(s)})
		}
		rows = append(rows, map[string]any{"name": name, "input": configInput(w), "config": configView(w), "error": message, "classes": classes, "default_state": w.DefaultState(), "status_names": w.StatusNames()})
	}
	add("default", DefaultWorkflowConfig())
	add("zero", WorkflowConfig{})
	add("empty", WorkflowConfig{Statuses: []string{}, Active: []string{}, Terminal: []string{}, Transitions: map[string][]string{}})
	for i, value := range values {
		w := DefaultWorkflowConfig()
		w.Statuses = append(w.Statuses, value)
		add(fmt.Sprintf("status-%d", i), w)
		w = DefaultWorkflowConfig()
		w.Statuses = append(w.Statuses, value, value)
		add(fmt.Sprintf("duplicate-%d", i), w)
		w = DefaultWorkflowConfig()
		w.Default = value
		add(fmt.Sprintf("default-%d", i), w)
		w = DefaultWorkflowConfig()
		w.Active = []string{value}
		add(fmt.Sprintf("active-%d", i), w)
		w = DefaultWorkflowConfig()
		w.Terminal = []string{value}
		add(fmt.Sprintf("terminal-%d", i), w)
		w = DefaultWorkflowConfig()
		w.Transitions = map[string][]string{value: {"open"}}
		add(fmt.Sprintf("source-%d", i), w)
		w = DefaultWorkflowConfig()
		w.Transitions = map[string][]string{"open": {value}}
		add(fmt.Sprintf("target-%d", i), w)
	}
	w := DefaultWorkflowConfig()
	w.Active = []string{"done", "closed", "done"}
	add("overlap-sorted-duplicates", w)
	w = DefaultWorkflowConfig()
	w.Statuses = []string{"open", "雪", "🌱"}
	w.Active = []string{"🌱", "雪"}
	w.Terminal = []string{"雪", "🌱"}
	add("unicode-overlap", w)
	w = DefaultWorkflowConfig()
	w.Transitions = map[string][]string{"open": nil}
	add("nil-transition-targets", w)
	files := []WorkflowFile{{}, {Statuses: []string{}, Active: []string{}, Terminal: []string{}, Transitions: map[string][]string{}}, {Default: " in_progress "}, {Statuses: []string{"open", "qa", "closed"}, Default: "open", Active: []string{"open"}, Terminal: []string{"closed"}}, {Transitions: map[string][]string{"open": nil}}}
	for _, value := range values {
		files = append(files, WorkflowFile{Statuses: []string{value}}, WorkflowFile{Default: value}, WorkflowFile{Active: []string{value}}, WorkflowFile{Terminal: []string{value}}, WorkflowFile{Transitions: map[string][]string{value: {value}}})
	}
	for i, file := range files {
		base := DefaultWorkflowConfig()
		base.Transitions = map[string][]string{"open": {"closed"}}
		merged := mergeWorkflowFile(base, workflowFile{Workflow: file})
		message := ""
		if e := merged.Validate(); e != nil {
			message = e.Error()
		}
		merges = append(merges, map[string]any{"name": fmt.Sprintf("merge-%d", i), "base": configInput(base), "file": configFileInput(file), "empty": file.IsEmpty(), "result": configView(merged), "error": message})
	}
	for i, names := range [][]string{nil, {}, DefaultTypes, {"story"}, {"task", "task"}, {""}, {" bug "}, {"雪", string([]byte{255})}} {
		cfg := TypesConfig{Names: names}
		checks := []any{}
		for _, value := range append(append([]string{}, DefaultTypes...), append([]string{"story"}, values...)...) {
			checks = append(checks, map[string]any{"name": configBytes(value), "valid": cfg.ValidType(value)})
		}
		types = append(types, map[string]any{"name": fmt.Sprintf("types-%d", i), "names": configList(names), "checks": checks})
	}
	inputs := []string{"", " ", "0", "+0", "-0", "00", "+", "-", ".s", "1", "1.", ".1", "1s", "1.s", ".5s", "-.5s", "5m", "not-a-duration", " 5m ", "\u20035m\u2003", "-1ns", "-0.1ns", "0.999999999ns", "1.999999999ns", "1s2m", "1m2s3ms4us5ns", "1s-2s", "1s+2s", "1e3s", "1d", "1µs", "1μs", "1μ", "1雪", "1\x00s", "雪", "1\x7fs", string([]byte{255}), "1\n", "1\\s", "1\"s", "9223372036854775807ns", "9223372036854775808ns", "-9223372036854775808ns", "-9223372036854775809ns", "9223372036854775808ns9223372036854775808ns", "-9223372036854775808ns9223372036854775808ns", "9223372036854775807ns1ns", "-9223372036854775807ns1ns", "9223372036854775807ns0.9ns", "2562047h47m16.854775807s", "-2562047h47m16.854775808s"}
	for _, unit := range []string{"ns", "us", "µs", "μs", "ms", "s", "m", "h", "bad"} {
		for _, value := range []string{"0", "1", ".0", ".99999999999999999999999999999", "1.00000000000000000000000000001", "9223372036854775807", "9223372036854775808", "9223372036854775809", "0000000000000000000000000000000000000.1", "0.000000000000000000000000000000001", "-1", "+1"} {
			inputs = append(inputs, value+unit)
		}
	}
	state := uint64(0x123456789abcdef0)
	units := []string{"ns", "us", "ms", "s", "m", "h"}
	for i := 0; i < 1024; i++ {
		state = state*6364136223846793005 + 1442695040888963407
		sign := ""
		if state&1 != 0 {
			sign = "-"
		}
		value := fmt.Sprintf("%s%d.%018d%s%dns", sign, (state>>8)%3000000, state%1000000000000000000, units[(state>>32)%6], state%1024)
		inputs = append(inputs, value)
	}
	inputs = append(inputs, "0."+strings.Repeat("0", 4096)+"1h", strings.Repeat("0", 4096)+"1s")
	for i, input := range inputs {
		d, e := time.ParseDuration(input)
		message := ""
		if e != nil {
			message = e.Error()
		}
		c := UserConfig{Fetch: UserFetchConfig{Throttle: input}}
		durations = append(durations, map[string]any{"name": fmt.Sprintf("duration-%d", i), "input": configBytes(input), "nanoseconds": int64(d), "error": message, "throttle": int64(c.ThrottleDuration())})
	}
	ranges := []any{}
	start := -1
	for r := 0; r <= 0x110000; r++ {
		printable := r < 0x110000 && unicode.IsPrint(rune(r))
		if printable && start < 0 {
			start = r
		}
		if !printable && start >= 0 {
			ranges = append(ranges, []int{start, r - 1})
			start = -1
		}
	}
	spaces := []any{}
	start = -1
	for r := 0; r <= 0x110000; r++ {
		space := r < 0x110000 && unicode.IsSpace(rune(r))
		if space && start < 0 {
			start = r
		}
		if !space && start >= 0 {
			spaces = append(spaces, []int{start, r - 1})
			start = -1
		}
	}
	out, e := json.Marshal(map[string]any{"workflows": rows, "merges": merges, "types": types, "durations": durations, "print_ranges": ranges, "unicode_version": unicode.Version, "space_ranges": spaces})
	if e != nil {
		t.Fatal(e)
	}
	if e = os.WriteFile(os.Getenv("BN_CONFIG_FOUNDATION_OUTPUT"), out, 0644); e != nil {
		t.Fatal(e)
	}
}
