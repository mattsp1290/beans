package vault
import("encoding/json";"os";"path/filepath";"strings";"testing";"regexp";"fmt")
func TestMigrationProjectFiles(t *testing.T) {
 type layout struct { Files map[string]string; Dirs []string; Links map[string]string }
 type node struct { Path []int; Kind string; Data []int; Mode uint32 }
 type input struct { Layout layout; Name,Remote []int }
 type stage struct { Paths [][]int; Error string; Tree []node }
 type row struct { Input input; Stages []stage }
 ints:=func(s string)[]int {v:=make([]int,len(s));for i:=range v {v[i]=int(s[i])};return v}
 str:=func(v []int)string {b:=make([]byte,len(v));for i,n:=range v {b[i]=byte(n)};return string(b)}
 snapshot:=func(root string)[]node {out:=[]node{};e:=filepath.WalkDir(root,func(path string,d os.DirEntry,e error)error {if e!=nil{return e};if path==root{return nil};info,e:=os.Lstat(path);if e!=nil{return e};n:=node{Path:ints(strings.TrimPrefix(path,root+"/")),Data:[]int{},Mode:uint32(info.Mode().Perm())};switch {case info.Mode()&os.ModeSymlink!=0:n.Kind="symlink";s,e:=os.Readlink(path);if e!=nil{return e};n.Data=ints(s);case info.IsDir():n.Kind="directory";default:n.Kind="file";b,e:=os.ReadFile(path);if e!=nil{return e};n.Data=ints(string(b))};out=append(out,n);return nil});if e!=nil{t.Fatal(e)};return out}
 layouts:=[]layout{{},{Files:map[string]string{"projects/p/beans.toml":"hand edited config"}},{Dirs:[]string{"projects/p/beans.toml"}},{Files:map[string]string{"projects":"blocking file"}},{Files:map[string]string{"projects/p":"blocking file"}},{Links:map[string]string{"projects/p":"absent"}},{Files:map[string]string{"target":"keep target"},Links:map[string]string{"projects/p/beans.toml":"../../target"}},{Dirs:[]string{"target"},Links:map[string]string{"projects/p/beans.toml":"../../target"}},{Links:map[string]string{"projects/p/beans.toml":"../../absent"}}}
 for _,sub:=range []string{"issues","archive","docs","memories","requests","handoffs","handoffs/archive"} {
  base:="projects/p/"+sub
  layouts=append(layouts,layout{Files:map[string]string{base+"/.gitkeep":"user data"}},layout{Dirs:[]string{base+"/.gitkeep"}},layout{Links:map[string]string{base+"/.gitkeep":"missing"}})
 }
 layouts=append(layouts,layout{Links:map[string]string{"projects/p/requests":"absent"}},layout{Links:map[string]string{"projects/p/requests":"../../target"},Dirs:[]string{"target"}},layout{Files:map[string]string{"projects/p/requests":"blocker"}})
 var inputs []input
 for _,l:=range layouts {for _,r:=range []string{"","https://github.com/o/p"} {inputs=append(inputs,input{Layout:l,Name:ints("p"),Remote:ints(r)})}}
 for _,n:=range []string{"雪","bad\xff","../other","","/absolute","nul\x00name"} {for _,r:=range []string{"","https://example.invalid/p","raw\xff","control\a\v\""} {inputs=append(inputs,input{Layout:layout{},Name:ints(n),Remote:ints(r)})}}
 var rows []row
 tempRe:=regexp.MustCompile(`\.bn-write-[0-9]+`)
 for i,in:=range inputs {root:=filepath.Join(t.TempDir(),fmt.Sprint(i));if e:=os.MkdirAll(root,0755);e!=nil{t.Fatal(e)}
  for _,p:=range in.Layout.Dirs {if e:=os.MkdirAll(filepath.Join(root,p),0755);e!=nil{t.Fatal(e)}}
  for p,s:=range in.Layout.Files {f:=filepath.Join(root,p);if e:=os.MkdirAll(filepath.Dir(f),0755);e!=nil{t.Fatal(e)};if e:=os.WriteFile(f,[]byte(s),0644);e!=nil{t.Fatal(e)}}
  for p,s:=range in.Layout.Links {f:=filepath.Join(root,p);if e:=os.MkdirAll(filepath.Dir(f),0755);e!=nil{t.Fatal(e)};if e:=os.Symlink(s,f);e!=nil{t.Fatal(e)}}
  out:=row{Input:in};for j:=0;j<2;j++ {paths,e:=CreateProjectFiles(root,str(in.Name),str(in.Remote));s:=stage{Tree:snapshot(root)};if paths!=nil {s.Paths=[][]int{};for _,p:=range paths{s.Paths=append(s.Paths,ints(p))}};if e!=nil{s.Error=tempRe.ReplaceAllString(strings.ReplaceAll(e.Error(),root+"/",""),".bn-write-TEMP")};out.Stages=append(out.Stages,s)};rows=append(rows,out)
 }
 b,e:=json.Marshal(map[string]any{"cases":rows});if e!=nil{t.Fatal(e)};if e=os.WriteFile(os.Getenv("BN_PROJECT_FILES_OUTPUT"),b,0600);e!=nil{t.Fatal(e)}
}
