package plan

import (
	"encoding/json"
	"fmt"
	"os"
	"strings"
	"testing"
	"time"
	"unicode/utf8"
)

func TestMigrationPlanParseCensus(t *testing.T) {
	rows := []map[string]any{}
	now := time.Date(2026, 9, 11, 12, 0, 0, 123456789, time.UTC)
	base := string(Scaffold("beans-plan-a3f2", "Add plan artifacts", now))
	view := func(p *Plan) any {
		if p == nil {
			return nil
		}
		stamp := func(t time.Time) any {
			return map[string]any{"seconds": t.Unix(), "nanoseconds": t.Nanosecond(), "offset_seconds": 0}
		}
		return map[string]any{"id": p.ID, "aliases": p.Aliases, "title": p.Title, "slug": p.Slug, "status": p.Status, "created": stamp(p.Created), "updated": stamp(p.Updated), "sections": p.Sections, "body": p.Body, "path": p.Path, "summary": map[string]any{"outcome": p.Summary.Outcome, "affected_areas": p.Summary.AffectedAreas, "execution_order": p.Summary.ExecutionOrder, "risks": p.Summary.Risks, "change_graph": p.Summary.ChangeGraph}, "graph": p.Graph}
	}
	add := func(name string, input []byte) {
		p, err := Parse("plan.md", input)
		message := ""
		if err != nil {
			message = err.Error()
		}
		row := map[string]any{"name": name, "input": string(input), "plan": view(p), "error": message}
		if p != nil {
			encoded, e := Encode(p)
			row["encoded"] = string(encoded)
			row["encode_error"] = ""
			if e != nil {
				row["encode_error"] = e.Error()
			}
		}
		if !utf8.Valid(input) {
			ints := []int{}
			for _, b := range input {
				ints = append(ints, int(b))
			}
			row["input_bytes"] = ints
		}
		rows = append(rows, row)
	}
	replace := func(key, value string) string {
		lines := strings.Split(base, "\n")
		for i, line := range lines {
			if strings.HasPrefix(line, key+":") {
				lines[i] = key + ": " + value
				for i+1 < len(lines) && strings.HasPrefix(lines[i+1], "    - ") {
					lines = append(lines[:i+1], lines[i+2:]...)
				}
				return strings.Join(lines, "\n")
			}
		}
		return strings.Replace(base, "updated:", key+": "+value+"\nupdated:", 1)
	}
	add("scaffold", []byte(base))
	add("empty", nil)
	add("nul", []byte("\x00"))
	add("utf8", []byte{255, 10})
	add("cr", []byte("\r\n"))
	add("no-final-lf", []byte(strings.TrimSuffix(base, "\n")))
	add("no-frontmatter", []byte("hello\n"))
	add("unclosed", []byte("---\nhello\n"))
	for _, fm := range []string{"", "null", "[]", "{}", "? [id]\n: x", "unknown: x"} {
		add("root-"+fm, []byte("---\n"+fm+"\n---\n"))
	}
	add("bad-key-indent", []byte(strings.Replace(base, "aliases:", "aliases: []", 1)))
	for _, key := range []string{"id", "aliases", "title", "slug", "status", "created", "updated", "sections"} {
		lines := strings.Split(base, "\n")
		for i, l := range lines {
			if strings.HasPrefix(l, key+":") {
				lines = append(lines[:i], lines[i+1:]...)
				break
			}
		}
		add("missing-"+key, []byte(strings.Join(lines, "\n")))
		add("duplicate-"+key, []byte(strings.Replace(base, "updated:", key+": null\nupdated:", 1)))
		for i, value := range []string{"null", "NULL", "~", "''", "' '", "true", "123", "[]", "{}", "[null]", "[true, 123, null]", "!!str null", "!beans x"} {
			add(fmt.Sprintf("%s-%d", key, i), []byte(replace(key, value)))
		}
	}
	for i, value := range []string{"bad", "", "null", "2026-09-11", "2026-09-11t12:00:00Z", "2026-09-11T12:00:00z", "2026-09-11T1:00:00Z", "2026-09-11T12:00:00,5Z", "2026-09-11T12:00:00.123456789123Z", "2026-09-11T12:00:00+00:00", "2026-09-11T12:00:00-00:00", "2026-09-11T12:00:00+24:00", "2026-09-11T12:00:00+00:60", "2026-09-11T12:00:00+25:61", "2026-13-11T12:00:00Z", "2026-00-11T12:00:00Z", "2026-02-30T12:00:00Z", "2026-02-00T12:00:00Z", "2026-02-99T12:00:00Z", "2026-09-11T24:00:00Z", "2026-09-11T12:60:00Z", "2026-09-11T12:00:60Z", "2026-09-11T12:00:00.Z", "2026-09-11T12:00:00Zextra", " 2026-09-11T12:00:00Z", "2026-09-11T12:00:00Z ", "0001-01-01T00:00:00Z", "0000-01-01T00:00:00Z", "雪", "2雪"} {
		encoded, _ := json.Marshal(value)
		add(fmt.Sprintf("time-%d", i), []byte(replace("created", string(encoded))))
	}
	for i, input := range []string{strings.Replace(base, "id: beans-plan-a3f2", "id: &id beans-plan-a3f2", 1), strings.Replace(strings.Replace(base, "id: beans-plan-a3f2", "id: &id beans-plan-a3f2", 1), "- beans-plan-a3f2", "- *id", 1), strings.Replace(base, "title: Add plan artifacts", "title: &title Add plan artifacts\nsections: [*title]", 1), strings.Replace(base, "title: Add plan artifacts", "title: *missing", 1), strings.Replace(base, "aliases:", "aliases: &aliases", 1) + "", strings.Replace(base, "title: Add plan artifacts", "title: &title Add plan artifacts\nsections: *title", 1)} {
		add(fmt.Sprintf("anchor-%d", i), []byte(input))
	}

	for i, value := range []string{"&a [beans-plan-a3f2]", "[*id]", "!!null []", "!!null {}", "[!!int x]", "[!!bool nope]", "[!!float 1]", "[!!timestamp x]", "[!!binary eA==]", "[!!binary '@']", "&a [*a]", "[!beans raw]", "[!!null x]"} {
		add(fmt.Sprintf("list-tag-%d", i), []byte(replace("aliases", value)))
	}
	for i, input := range []string{strings.Replace(replace("sections", "*a"), "aliases:", "aliases: &a", 1), strings.Replace(base, "title: Add plan artifacts", "title: &name Add plan artifacts\nsections: [*name, *name]", 1), strings.Replace(base, "title: Add plan artifacts", "title: &x [name]\nsections: [*x]", 1)} {
		add(fmt.Sprintf("alias-list-%d", i), []byte(input))
	}
	valid := "2026-09-11T12:00:00Z"
	for n := 0; n < len(valid); n++ {
		v, _ := json.Marshal(valid[:n])
		add(fmt.Sprintf("time-truncated-%d", n), []byte(replace("created", string(v))))
	}
	for i, value := range []string{"2026-09-11T12:00:00Z\n", "2026-09-11T12:00:00Z雪", "2026-09-11T12:00:00Z\x7f", "2026-09-11T12:00:00*25:00", "2026-09-11T12:00:00+ab:61", "2026-09-11T12:00:00+25:ab", "2024-02-29T00:00:00Z", "1900-02-29T00:00:00Z", "2000-02-29T00:00:00Z", "0001-01-01T00:00:00.1Z"} {
		v, _ := json.Marshal(value)
		add(fmt.Sprintf("time-edge-%d", i), []byte(replace("created", string(v))))
	}

	for i, value := range []string{"eA==", "/w==", "4oI=", "Cgo=", "wqE=", "", "AA==", "Zh==", "Zm9=", "Zm9v", "Zg", "Zg=", "Zg===", "Zg==Zg==", "Z g==", "===="} {
		add(fmt.Sprintf("binary-%d", i), []byte(replace("aliases", "[beans-plan-a3f2, !!binary '"+value+"']")))
	}

	scaffolds := []map[string]any{}
	for i, title := range []string{"", "x", "Ship: phase #1 \"quoted\"", "雪", "🌱", "null", "a\nb", " a\nb", "a\x00b", "a\r\nb", strings.Repeat("a", 61), "!!!", "\n", "\n\n"} {
		id := "beans-plan-a3f2"
		data := Scaffold(id, title, now)
		scaffolds = append(scaffolds, map[string]any{"id": id, "title": title, "now": map[string]any{"seconds": now.Unix(), "nanoseconds": now.Nanosecond(), "offset_seconds": 0}, "output": string(data)})
		add(fmt.Sprintf("scaffold-title-%d", i), data)
	}

	for i, value := range []string{"{x: a, x: b}", "[{x: a, x: b}, []]", "[[], [], {}]", "[!!int x, []]", "[[], !!int x]"} {
		add(fmt.Sprintf("list-error-order-%d", i), []byte(replace("sections", value)))
	}
	add("comments-only", []byte("---\n# comment\n---\n"))

	end := strings.Index(base[4:], "\n---\n") + 4
	fm := base[4:end]
	body := base[end+5:]
	for _, position := range []int{500, 509, 510, 511, 512, 513, 1023, 1024} {
		prefix := fm + "\n...\n#"
		pad := position - len(prefix)
		add(fmt.Sprintf("unread-control-%d", position), []byte("---\n"+prefix+strings.Repeat("a", pad)+"\x7f\n---\n"+body))
	}

	for i, tail := range []string{"a: b", "[]", "[a,b,c,d,e]", "a", "# only comment"} {
		prefix := fm + "\n...\n" + tail + "\n#"
		add(fmt.Sprintf("document-tail-%d", i), []byte("---\n"+prefix+strings.Repeat("a", 1024-len(prefix))+"\x7f\n---\n"+body))
	}
	out, err := json.Marshal(map[string]any{"parses": rows, "scaffolds": scaffolds})
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(os.Getenv("BN_PLAN_PARSE_OUTPUT"), out, 0644); err != nil {
		t.Fatal(err)
	}
}
