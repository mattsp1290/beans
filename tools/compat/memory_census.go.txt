package issue

import (
    "encoding/json"
    "os"
    "testing"
    "time"
    "gopkg.in/yaml.v3"
)

func memoryMigrationMetadata(m *Memory) map[string]any {
    stamp:=func(v time.Time)map[string]any { _,offset:=v.Zone();return map[string]any{"seconds":v.Unix(),"nanoseconds":v.Nanosecond(),"offset_seconds":offset} }
    tags:=m.Tags;if tags==nil { tags=[]string{} }
    return map[string]any{"key":m.Key,"type":m.Type,"tags":tags,"created":stamp(m.Created),"updated":stamp(m.Updated),"project":m.Project}
}
func TestMigrationMemoryCensus(t *testing.T) {
    raw,err:=os.ReadFile(os.Getenv("BN_MEMORY_INPUT"));if err!=nil { t.Fatal(err) }
    var cases []map[string]any;if err:=json.Unmarshal(raw,&cases);err!=nil { t.Fatal(err) }
    for _,c:=range cases {
        path,input:=c["path"].(string),c["input"].(string)
        m,err:=ParseMemory(path,[]byte(input));if err!=nil { c["error"]=err.Error();continue }
        c["metadata"]=memoryMigrationMetadata(m);c["body"]=m.Body
        c["valid_key"]=ValidMemoryKey(m.Key);c["valid_type"]=ValidMemoryType(m.Type)
        unknown:=[]string{};for i:=0;i+1<len(m.Extra.Content);i+=2 { unknown=append(unknown,m.Extra.Content[i].Value) };c["unknown"]=unknown
        out,err:=EncodeMemory(m);if err!=nil { t.Fatal(err) };c["encoded"]=string(out)
        if os.Getenv("BN_MEMORY_READ_ONLY")=="1" { continue }
        edits:=[]map[string]any{}
        for _,kind:=range []string{"key","key-empty","type","tags","created","updated","clear-optionals","body","all","same-instant","new","new-extra"} {
            changed,_:=ParseMemory(path,[]byte(input))
            apply:=func(key string) {
                switch key {
                case "key":changed.Key="New Key 雪"
                case "key-empty":changed.Key=""
                case "type":changed.Type="reference"
                case "tags":changed.Tags=[]string{"true","snow 雪"}
                case "created":changed.Created=time.Date(2027,1,2,3,4,5,123000000,time.FixedZone("offset",19800))
                case "updated":changed.Updated=time.Date(2027,1,2,3,4,5,123000000,time.UTC)
                case "clear-optionals":changed.Type="";changed.Tags=nil;changed.Created=time.Time{};changed.Updated=time.Time{}
                case "body":changed.Body="replacement\n## Log\nopaque\n"
                case "same-instant":changed.Created=changed.Created.UTC();changed.Updated=changed.Updated.UTC()
                }
            }
            if kind=="all" { for _,key:=range []string{"key","type","tags","created","updated","body"} { apply(key) } } else if kind=="new" || kind=="new-extra" {
                changed.orig=nil;changed.Extra=yaml.Node{}
                if kind=="new-extra" { changed.Extra=yaml.Node{Kind:yaml.MappingNode,Tag:"!!map",Content:[]*yaml.Node{keyNode("custom"),strNode("雪: value")}} }
            } else { apply(kind) }
            out,err:=EncodeMemory(changed);if err!=nil { t.Fatal(err) }
            edits=append(edits,map[string]any{"kind":kind,"encoded":string(out)})
        }
        c["edits"]=edits
    }
    out,err:=json.Marshal(cases);if err!=nil { t.Fatal(err) }
    if err:=os.WriteFile(os.Getenv("BN_MEMORY_OUTPUT"),out,0644);err!=nil { t.Fatal(err) }
}
