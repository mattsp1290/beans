package issue

import (
    "encoding/json"
    "os"
    "testing"
    "time"
    "gopkg.in/yaml.v3"
)

func requestMigrationMetadata(r *Request) map[string]any {
    stamp := func(v time.Time) map[string]any { _, offset := v.Zone(); return map[string]any{"seconds":v.Unix(), "nanoseconds":v.Nanosecond(), "offset_seconds":offset} }
    list := func(v []string) []string { if v == nil { return []string{} }; return v }
    links := []map[string]string{}
    for _, l := range r.Issues { links = append(links, map[string]string{"raw":l.Raw,"target":l.Target}) }
    return map[string]any{"id":r.ID,"title":r.Title,"status":r.Status,"priority":r.Priority,"labels":list(r.Labels),"requested_by":r.RequestedBy,"issues":links,"aliases":list(r.Aliases),"created":stamp(r.Created),"updated":stamp(r.Updated),"project":r.Project}
}

func requestMigrationLog(entries []LogEntry) []map[string]any {
    out := []map[string]any{}
    for _, e := range entries {
        _, offset := e.At.Zone()
        out = append(out,map[string]any{"at":map[string]any{"seconds":e.At.Unix(),"nanoseconds":e.At.Nanosecond(),"offset_seconds":offset},"actor":e.Actor,"repo":e.Repo,"sha":e.SHA,"branch":e.Branch,"event":e.Event,"raw":e.Raw})
    }
    return out
}

func TestMigrationRequestCensus(t *testing.T) {
    raw, err := os.ReadFile(os.Getenv("BN_REQUEST_INPUT")); if err != nil { t.Fatal(err) }
    var cases []map[string]any
    if err := json.Unmarshal(raw, &cases); err != nil { t.Fatal(err) }
    for _, c := range cases {
        path, input := c["path"].(string), c["input"].(string)
        r, err := ParseRequest(path, []byte(input))
        if err != nil { c["error"] = err.Error(); continue }
        c["metadata"] = requestMigrationMetadata(r)
        c["body"] = r.Body
        c["log"] = requestMigrationLog(r.Log)
        unknown := []string{}
        for i:=0;i+1<len(r.Extra.Content);i+=2 { unknown=append(unknown,r.Extra.Content[i].Value) }
        c["unknown"] = unknown
        output, err := EncodeRequest(r); if err != nil { t.Fatal(err) }; c["encoded"] = string(output)
        if os.Getenv("BN_REQUEST_READ_ONLY") == "1" { continue }
        edits := []map[string]any{}
        for _, kind := range []string{"id","aliases","title","status","priority","labels","requested_by","issues","created","updated","body","body-empty","append","opaque-log","all","new","new-extra","clear-optionals","invalid-status","invalid-id","invalid-priority","blank-title","zero-created","equal-log-time","same-instant"} {
            changed, _ := ParseRequest(path, []byte(input))
            apply := func(key string) {
                switch key {
                case "id": changed.ID = "foreign-r-b4c5.1"
                case "aliases": changed.Aliases = []string{"external"}
                case "title": changed.Title = "雪: new title"
                case "status": changed.Status = RequestAccepted
                case "priority": changed.Priority = 4
                case "labels": changed.Labels = []string{"true","snow 雪"}
                case "requested_by": changed.RequestedBy = "New Actor"
                case "issues": changed.Issues = []Link{ParseLink("[[other-abc#heading|alias]]"),ParseLink("bare")}
                case "created": changed.Created = time.Date(2027,1,2,3,4,5,123000000,time.FixedZone("offset",19800))
                case "updated": changed.Updated = time.Date(2027,1,2,3,4,5,123000000,time.UTC)
                case "body": SetRequestBody(changed,"replacement\n\n")
                case "body-empty": SetRequestBody(changed,"")
                case "append": AppendRequestLog(changed,LogEntry{At:time.Date(2028,1,2,3,4,5,0,time.UTC),Actor:"matt",Event:"status open → accepted\ncontinued"})
                case "clear-optionals": changed.Labels=nil; changed.RequestedBy=""; changed.Issues=nil
                case "invalid-status": changed.Status="closed"
                case "invalid-id": changed.ID="beans-a3f2"
                case "invalid-priority": changed.Priority=5
                case "blank-title": changed.Title=" \t\n"
                case "zero-created": changed.Created=time.Time{}
                case "equal-log-time": AppendRequestLog(changed,LogEntry{At:changed.Updated, Actor:"matt",Event:"equal time"})
                case "same-instant": changed.Created=changed.Created.UTC(); changed.Updated=changed.Updated.UTC()
                case "opaque-log": if len(changed.Log)>0 { changed.Log[0].Raw="- replaced original"; changed.Log[0].Event="ignored" }
                }
            }
            if kind == "all" { for _, key := range []string{"title","status","priority","labels","requested_by","issues","body","append"} { apply(key) } } else if kind == "new" || kind == "new-extra" {
                changed.orig=nil; changed.Extra=yaml.Node{}
                if kind == "new-extra" { changed.Extra=yaml.Node{Kind:yaml.MappingNode,Tag:"!!map",Content:[]*yaml.Node{keyNode("custom"),strNode("雪: value"),keyNode("nested"),flowSeq([]string{"true","other"})}} }
            } else { apply(kind) }
            encoded, err := EncodeRequest(changed)
            edit := map[string]any{"kind":kind}
            if err != nil { edit["error"]=err.Error() } else { edit["encoded"]=string(encoded) }
            edits=append(edits,edit)
        }
        c["edits"] = edits
    }
    output, err := json.Marshal(cases); if err != nil { t.Fatal(err) }
    if err := os.WriteFile(os.Getenv("BN_REQUEST_OUTPUT"),output,0644); err != nil { t.Fatal(err) }
}
