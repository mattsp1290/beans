package plan

import (
    "encoding/json"
    "fmt"
    "os"
    "strings"
    "testing"
    "gopkg.in/yaml.v3"
)

func TestMigrationPlanGraphCensus(t *testing.T) {
    graphs:=[]map[string]any{}
    fence:=func(yaml string)string { return "```bn-change-graph\n"+yaml+"\n```" }
    add:=func(name,text string) {
        g,err:=ParseGraph("plan.md",text);message:="";if err!=nil { message=err.Error() }
        graphs=append(graphs,map[string]any{"name":name,"input":text,"graph":g,"error":message})
    }
    add("missing","");add("unterminated","```bn-change-graph\nversion: 1");add("multiple",fence("version: 1")+"\n"+fence("version: 1"));add("prose","prose\n"+fence("version: 1"))
    for i,yaml:=range []string{"", "[]", "null", "{version: 1, nodes: [], edges: []}","version: 1", "version: 1\nnodes: null\nedges: null", "version: 1\nnodes: &nodes []\nedges: *nodes", "version: 1\nedges: *missing", "version: !beans 1", "version: 1\ncolor: red", "version: 1\nversion: 1", "!beans {version: 1}", "version: 1\nnodes: 1", "version: 1\nedges: {}", "version: 1\nnodes: [null]", "version: 1\nnodes: [{id: a, label: A, kind: component, color: red}]", "version: 1\nedges: [{from: a, to: a, kind: precedes, color: red}]"} { add(fmt.Sprintf("shape-%d",i),fence(yaml)) }
    for i,value:=range []string{"0", "2", "null", "'1'", "!!str 1", "0x1", "01", "+1", "true", "1.0", "2026-09-10", "!!int '1'", "!!int x", "! 1", "9223372036854775808", "'ab雪雪雪'", "'a🌱🌱🌱'", "'雪雪雪雪'"} { add(fmt.Sprintf("version-%d",i),fence("version: "+value+"\nnodes: []\nedges: []")) }
    base:=ChangeGraph{Version:1,Nodes:[]GraphNode{{ID:"a",Label:"A",Kind:"component"},{ID:"b",Label:"B",Kind:"workflow"}},Edges:[]GraphEdge{{From:"a",To:"b",Kind:"precedes"}}}
    generated:=func(name string,g ChangeGraph) { data,err:=yaml.Marshal(g);if err!=nil { t.Fatal(err) };add(name,fence(string(data))) }
    generated("valid",base)
    for i,id:=range []string{"", "A", "1a", "a--", "a-", "a.1", strings.Repeat("a",64),strings.Repeat("a",65),"雪"} { g:=base;g.Nodes=append([]GraphNode(nil),base.Nodes...);g.Nodes[0].ID=id;g.Edges=nil;generated(fmt.Sprintf("id-%d",i),g) }
    for i,label:=range []string{"", " ", "a\nb", "a\rb",strings.Repeat("a",120),strings.Repeat("a",121),strings.Repeat("雪",40),strings.Repeat("雪",41)} { g:=base;g.Nodes=append([]GraphNode(nil),base.Nodes...);g.Nodes[0].Label=label;generated(fmt.Sprintf("label-%d",i),g) }
    for _,kind:=range []string{"artifact","component","interface","data","workflow","external","unknown"} { g:=base;g.Nodes=append([]GraphNode(nil),base.Nodes...);g.Nodes[0].Kind=kind;generated("node-kind-"+kind,g) }
    for _,kind:=range []string{"precedes","affects","enables","produces","replaces","contains","unknown"} { g:=base;g.Edges=append([]GraphEdge(nil),base.Edges...);g.Edges[0].Kind=kind;generated("edge-kind-"+kind,g) }
    g:=base;g.Nodes=[]GraphNode{base.Nodes[0],base.Nodes[0]};generated("duplicate-node",g)
    g=base;g.Edges=[]GraphEdge{base.Edges[0],base.Edges[0]};generated("duplicate-edge",g)
    g=base;g.Edges=[]GraphEdge{{From:"missing",To:"b",Kind:"precedes"}};generated("missing-endpoint",g)
    for _,count:=range []int{200,201} { g:=ChangeGraph{Version:1};for i:=0;i<count;i++ { g.Nodes=append(g.Nodes,GraphNode{ID:fmt.Sprintf("n%d",i),Label:"N",Kind:"data"}) };generated(fmt.Sprintf("nodes-%d",count),g) }
    for _,count:=range []int{400,401} { g:=base;g.Edges=nil;for i:=0;i<count;i++ { g.Edges=append(g.Edges,GraphEdge{From:"a",To:"b",Kind:"affects",Label:fmt.Sprintf("e%d",i)}) };generated(fmt.Sprintf("edges-%d",count),g) }
    for i,value:=range []string{"true", "1.0", "2026-09-10", "!beans A", "!!str true", "null", "123", "[a, b]", "{a: b}"} { add(fmt.Sprintf("label-type-%d",i),fence("version: 1\nnodes: [{id: a, label: "+value+", kind: component}]\nedges: []")) }
    add("nested-duplicate",fence("version: 1\nnodes: [{id: a, id: b, label: A, kind: component}]"))
    add("merge",fence("version: 1\nnodes: [{<<: {}, id: a, label: A, kind: component}]"))
    add("comments-only",fence("# comment"))
    for i,yaml:=range []string{"version: !!null []", "version: 1\nnodes: !!null []\nedges: !!null []", "version: 1\nnodes: !!null {}", "version: 1\nnodes: [{id: a, label: !!null {}, kind: component}]"}{add(fmt.Sprintf("null-container-%d",i),fence(yaml))}
    summaries:=[]map[string]any{}
    body:="## Summary\n\n### Outcome\n```text\n## not a section\n```\n\n### Affected areas\n- x\n\n### Execution order\n1. x\n\n### Risks\n- none\n\n### Change graph\n"+fence("version: 1\nnodes: []\nedges: []")+"\n\n## Next\n"
    inputs:=[]string{body,"",body+"## Summary\n",strings.Replace(body,"### Outcome","### Risks",1),strings.Replace(body,"### Risks","### Outcome",1),strings.Replace(body,"### Outcome","  ### Outcome",1),strings.Replace(body,"## Summary","  ## Summary  ",1)}
    for _,name:=range []string{"Outcome","Affected areas","Execution order","Risks","Change graph"} { inputs=append(inputs,strings.Replace(body,"### "+name,"### Unknown",1)) }
    inputs=append(inputs,strings.Replace(body,"### Outcome","### Outcome\n### Outcome",1),strings.Replace(body,"version: 1","version: 2",1))
    for i,input:=range inputs { summary,graph,err:=parseSummary("plan.md",input);message:="";if err!=nil { message=err.Error() };summaries=append(summaries,map[string]any{"name":fmt.Sprintf("summary-%d",i),"input":input,"summary":map[string]string{"outcome":summary.Outcome,"affected_areas":summary.AffectedAreas,"execution_order":summary.ExecutionOrder,"risks":summary.Risks,"change_graph":summary.ChangeGraph},"graph":graph,"error":message}) }
    output,err:=json.Marshal(map[string]any{"graphs":graphs,"summaries":summaries});if err!=nil { t.Fatal(err) }
    if err:=os.WriteFile(os.Getenv("BN_PLAN_GRAPH_OUTPUT"),output,0644);err!=nil { t.Fatal(err) }
}
