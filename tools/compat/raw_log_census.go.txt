func readerInts(b []byte) []int {
	v := make([]int, len(b))
	for i, b := range b {
		v[i] = int(b)
	}
	return v
}
func TestMigrationRawLogProbe(t *testing.T) {
	rows := []any{}
	inputs := map[string]any{}
	sequences := [][]byte{}
	for b := 0; b < 256; b++ {
		sequences = append(sequences, []byte{byte(b)})
	}
	for _, s := range []string{" \t\n\r\v\f", "\u0085\u00a0\u1680\u2000\u2028\u202f\u3000", "\u200b\ufeff", "\xe2\x82", "\xc0\xaf", "\xed\xa0\x80", "\xf4\x90\x80\x80", "\xff\n\r\nend"} {
		sequences = append(sequences, []byte(s))
	}
	for _, field := range []string{"actor", "repo", "sha", "branch", "event", "raw"} {
		for i, sequence := range sequences {
			e := LogEntry{At: time.Date(2027, 1, 1, 0, 0, 0, 0, time.UTC), Actor: "actor", Repo: "repo", SHA: "abcdef", Branch: "branch", Event: "event"}
			value := "A " + string(sequence) + " Z"
			switch field {
			case "actor":
				e.Actor = value
			case "repo":
				e.Repo = value
			case "sha":
				e.SHA = value
			case "branch":
				e.Branch = value
			case "event":
				e.Event = value
			case "raw":
				e.Raw = value
			}
			formatted := FormatLogEntry(e)
			line := logLine(e)
			parsed, ok := ParseLogEntry(strings.TrimSuffix(line, "\n"))
			entry := map[string]any{"ok": ok}
			if ok {
				entry["actor"] = readerInts([]byte(parsed.Actor))
				entry["repo"] = readerInts([]byte(parsed.Repo))
				entry["sha"] = readerInts([]byte(parsed.SHA))
				entry["branch"] = readerInts([]byte(parsed.Branch))
				entry["event"] = readerInts([]byte(parsed.Event))
			}
			outputs := map[string]any{}
			for _, kind := range []string{"issue", "request"} {
				path := "projects/p/issues/p-one.md"
				head := "id: p-one\ntitle: Title\ntype: task\nstatus: open\npriority: 2\n"
				if kind == "request" {
					path = "projects/p/requests/p-r-one.md"
					head = "id: p-r-one\naliases: [p-r-one]\ntitle: Title\nstatus: open\npriority: 2\n"
				}
				for _, bodyKind := range []string{"log", "no-log", "empty", "fence", "new"} {
					body := "prefix\xff\n\n## Log\n- opaque\xff\n\n## Tail\ntail\xff\n"
					switch bodyKind {
					case "no-log":
						body = "prefix\xff\n\n## Tail\ntail\xff\n"
					case "empty", "new":
						body = ""
					case "fence":
						body = "prefix\xff\n---\n"
					}
					input := []byte("---\n" + head + "created: 2026-01-01T00:00:00Z\nupdated: 2026-01-01T00:00:00Z\n---\n" + body)
					inputs[kind+":"+bodyKind] = map[string]any{"path": path, "bytes": readerInts(input)}

					var out []byte
					var err error
					if kind == "issue" {
						d, e2 := Parse(path, input)
						if e2 != nil {
							t.Fatal(e2)
						}
						if bodyKind == "new" {
							d.orig = nil
						}
						AppendLog(d, e)
						out, err = Encode(d)
					} else {
						d, e2 := ParseRequest(path, input)
						if e2 != nil {
							t.Fatal(e2)
						}
						if bodyKind == "new" {
							d.orig = nil
						}
						AppendRequestLog(d, e)
						out, err = EncodeRequest(d)
					}
					if err != nil {
						t.Fatal(err)
					}
					reread, readError := readerRead(kind, path, out)
					message := ""
					if readError != nil {
						message = readError.Error()
					} else if !bytes.Equal(reread, out) {
						t.Fatal("raw log reread changes bytes")
					}
					outputs[kind+":"+bodyKind] = map[string]any{"bytes": readerInts(out), "reread_error": message}
				}
			}
			rows = append(rows, map[string]any{"field": field, "sequence_index": i, "sequence": readerInts(sequence), "formatted": readerInts([]byte(formatted)), "line": readerInts([]byte(line)), "parsed": entry, "outputs": outputs})
		}
	}
	data, err := json.Marshal(map[string]any{"cases": rows, "inputs": inputs})
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(os.Getenv("BN_RAW_LOG_OUTPUT"), data, 0600); err != nil {
		t.Fatal(err)
	}
	fmt.Println("raw log cases", len(rows))
}

func readerRead(kind, path string, input []byte) ([]byte, error) {
	switch kind {
	case "issue":
		d, e := Parse(path, input)
		if e != nil {
			return nil, e
		}
		return Encode(d)
	case "request":
		d, e := ParseRequest(path, input)
		if e != nil {
			return nil, e
		}
		return EncodeRequest(d)
	case "memory":
		d, e := ParseMemory(path, input)
		if e != nil {
			return nil, e
		}
		return EncodeMemory(d)
	case "handoff":
		d, e := ParseHandoff(path, input)
		if e != nil {
			return nil, e
		}
		return EncodeHandoff(d)
	}
	panic("unknown kind")
}
func TestMigrationRawReaderRead(t *testing.T) {
	raw, e := os.ReadFile(os.Getenv("BN_RUST_RAW_READER_INPUT"))
	if e != nil {
		t.Fatal(e)
	}
	var rows []struct {
		Kind, Path, ExpectedError string
		Bytes                     []int
	}
	if e := json.Unmarshal(raw, &rows); e != nil {
		t.Fatal(e)
	}
	accepted, rejected := 0, 0
	for i, r := range rows {
		input := make([]byte, len(r.Bytes))
		for j, b := range r.Bytes {
			input[j] = byte(b)
		}
		output, e := readerRead(r.Kind, r.Path, input)
		message := ""
		if e != nil {
			message = e.Error()
		}
		if message != r.ExpectedError {
			t.Fatalf("%d: got %s want %s", i, message, r.ExpectedError)
		}
		if e == nil {
			accepted++
			if !bytes.Equal(input, output) {
				t.Fatalf("Go rewrites Rust output %d", i)
			}
		} else {
			rejected++
		}
	}
	t.Logf("Go read %d actual Rust outputs: %d accepted, %d matching rejections", len(rows), accepted, rejected)
}
